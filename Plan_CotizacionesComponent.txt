                vm.getRenegotiationType = function(producto, opportunity, cotizacion) {//CEAM: 3563
                    var alza = 0, baja = 0;
                    if(!producto || Object.keys(producto).length < 1) {return;}
                    var renegotiationType = '';
                    var modalidadPago = '';
                    angular.forEach(Object.keys(producto), function (conceptoKey) {
                        var concepto = producto[conceptoKey]
                        if (concepto.Render &&
                            (concepto.TipoCargo__c == 'Comisión por el servicio' ||
                            concepto.TipoCargo__c == 'Comisión por servicio piloto' ||
                            concepto.TipoCargo__c == 'Comisión por Servicio Telepeaje' ||

                            concepto.TipoCargo__c == 'Cargo por envío - Recurrente' ||
                            concepto.TipoCargo__c == 'Cargo por envío estándar' ||
                            concepto.TipoCargo__c == 'Cargo por envío' ||
                            concepto.TipoCargo__c == 'Tarjeta titular / Stock' ||
                            concepto.TipoCargo__c == 'Tarjeta titular')
                            ) {
                            if(concepto.PrecioLista__c && concepto.PrecioFinal__c != concepto.PrecioLista__c) {
                                concepto.PrecioFinal__c > concepto.PrecioLista__c ? alza += 1 : baja += 1;
                            }
                            if(concepto.PrecioFinal__c && concepto.PrecioFinal__c != concepto.Importe__c) {
                                concepto.PrecioFinal__c > concepto.Importe__c ? alza += 1 : baja += 1;
                            }
                            if(concepto.PrecioFinal2__c && concepto.PrecioFinal2__c != concepto.Importe__c) {
                                concepto.PrecioFinal2__c > concepto.Importe__c ? alza += 1 : baja += 1;
                            }
                        }
                    })

                    if(typeof cotizacion.modalidadPago == 'undefined') {
                        cotizacion.productos[0].PaymentMethod__c = null;
                        modalidadPago = '';
                    } else if(cotizacion.modalidadPago) {
                        modalidadPago = cotizacion.productos[0].PaymentMethod__c;
                    }

                    renegotiationType = alza > 0 && baja < 1 ? 'Alza' : alza < 1 && baja > 0 ? 'Baja' : alza > 0 && baja > 0 ? 'AlzaBaja' : '';
                    renegotiationType = !!renegotiationType ? renegotiationType.concat(modalidadPago) : modalidadPago;
                    console.log('RenegotiationType Return',renegotiationType);
                    return renegotiationType;
