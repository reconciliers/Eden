    private static void ValidateOpportunityWithRFC(List<Opportunity> newList, Map<Id, Opportunity> oldMap) {
        Set<String> solutionList = new Set<String>();
        Set<Id> entidadList = new Set<Id>();
        Set<Id> accountList = new Set<Id>();
        Set<Id> oppsId = new Set<Id>();
        Set<String> productList = new Set<String>();
        List<Opportunity> oppsToValidate = new List<Opportunity>();
        for(Opportunity newOpp : newList){
            /* Modificaciones Televia */
            if(!newOpp.NTT_IsMigratedOpp__c){
                if(String.isNotBlank(newOpp.Entidad_Cuenta__c) && String.isNotBlank(newOpp.Familia__c) && String.isNotBlank(newOpp.Producto_cotizacion__c)) {
                    solutionList.add(newOpp.Familia__c);
                    entidadList.add(newOpp.Entidad_Cuenta__c);
                    accountList.add(newOpp.AccountId);
                    productList.add(newOpp.Producto_cotizacion__c);
                    oppsToValidate.add(newOpp);
                    oppsId.add(newOpp.Id);
                }
            }
        }
        if (!oppsToValidate.isEmpty()) {
            //validar duplicado
            List<Opportunity> existingOpps = [
                Select AccountId, Entidad_Cuenta__c, Familia__c, Producto_cotizacion__c, RecordTypeId
                from Opportunity
                where StageName Not In ('Ganada', 'Perdida', 'Planeada', 'Renegociada')
                and Modify_opportunity__c = true
                AND Familia__c IN: solutionList AND AccountId IN:accountList AND Entidad_Cuenta__c IN: entidadList
                AND Producto_cotizacion__c IN: productList
                AND Id NOT IN: oppsId
            ];
            if(!existingOpps.isEmpty()){
                for(Opportunity opp : oppsToValidate){
                    Opportunity oldOpp = oldMap.get(opp.Id);
                    //ya fue ingresado pero se cambia el producto
                    if(oldOpp.Entidad_Cuenta__c  != opp.Entidad_Cuenta__c || oldOpp.Producto_cotizacion__c  != opp.Producto_cotizacion__c){
                        for(Opportunity existingOpp : existingOpps){
                            if(opp.AccountId == existingOpp.AccountId && opp.Familia__c == existingOpp.Familia__c && opp.Producto_cotizacion__c == existingOpp.Producto_cotizacion__c && opp.Entidad_Cuenta__c == existingOpp.Entidad_Cuenta__c && opp.RecordTypeId == existingOpp.RecordTypeId){
                                opp.AddError('Ya existe una oportunidad abierta para ese mismo RFC Solución.');
                            }
                        }
                    }
                }
            }
        }
    }

    public static void ValidatepadlockOpportunity(List<Opportunity> newList){
        //Obtener recortypes
        Id incrementoRrcdt = Schema.SObjectType.Opportunity.getRecordTypeInfosByName().get('Upselling (Incremento)').getRecordTypeId();
        Set<String> solutionList = new Set<String>();
        Set<Id> entidadList = new Set<Id>();
        Set<Id> accountList = new Set<Id>();
        List<Opportunity> oppsToValidate = new List<Opportunity>();
        for(Opportunity newOpp : newList){
            /* Modificaciones Televia */
            if(!newOpp.NTT_IsMigratedOpp__c){
                if(newOpp.StageName == 'Planeada' && newOpp.RecordTypeId == incrementoRrcdt && String.isNotBlank(newOpp.Entidad_Cuenta__c) && String.isNotBlank(newOpp.Familia__c)) {
                    solutionList.add(newOpp.Familia__c);
                    entidadList.add(newOpp.Entidad_Cuenta__c);
                    accountList.add(newOpp.AccountId);
                    oppsToValidate.add(newOpp);
                }
            }/* Modificaciones Televia */
        }
        if(!oppsToValidate.isEmpty()){
            List<Opportunity> existingOpps = [
                SELECT AccountId, Entidad_Cuenta__c, Familia__c, RecordTypeId
                FROM Opportunity
                WHERE StageName Not In ('Ganada', 'Perdida', 'Renegociada')
                AND RecordTypeId =: incrementoRrcdt
                AND AccountId IN: accountList
                AND Entidad_Cuenta__c IN: entidadList
                AND Familia__c IN: solutionList
            ];
            if (!existingOpps.isEmpty()) {
                for(Opportunity newOpp : oppsToValidate){
                    for(Opportunity existingOpp : existingOpps){
                        if(newOpp.AccountId == existingOpp.AccountId && newOpp.Familia__c == existingOpp.Familia__c && newOpp.Entidad_Cuenta__c == existingOpp.Entidad_Cuenta__c && newOpp.RecordTypeId == existingOpp.RecordTypeId){
                            newOpp.AddError('Ya existe una oportunidad abierta para ese mismo RFC Solución.');
                        }
                    }
                }
            }
        }
    }
    public void tipificarOportunidad(){
        Set<Id> idOpps = new Set<Id>();
        Set<Id> entidadesCuenta = new Set<Id>();
        String accId = '';
        for(Opportunity opp : this.newList){
            /* Modificaciones Televia */
            if(!opp.NTT_IsMigratedOpp__c){
                if(isRecordType('VentaNueva', opp.RecordTypeId) || isRecordType('Migracion', opp.RecordTypeId) || isRecordType('BPfleet', opp.RecordTypeId)){
                    idOpps.add(opp.Id);
                    accId = opp.AccountId;
                    if(opp.Entidad_Cuenta__c != null){
                        entidadesCuenta.add(opp.Entidad_Cuenta__c);
                    }
                }
            }
        }
        Map<Id, List<ContratoFilial__c>> mapaFilial = new Map<Id, List<ContratoFilial__c>>();
        /*modificacion Televia add !String.isBlank(accId)*/
        if( !String.isBlank(accId) && accId != null && !mapaFilial.isEmpty() ){
            //Cambio Vero Zarza ticket soporte se agrega campo CreatedDate
            for(ContratoFilial__c ec : [Select Id, CodigoAS400__c, Entidad_Cuenta__c, Producto__r.Family, Contrato_2__c, Contrato_2__r.PartidaPresupuesto__c, Contrato_2__r.PartidaPresupuesto__r.Quote.OpportunityId, CreatedDate FROM ContratoFilial__c
                                        WHERE Entidad_Cuenta__c IN (SELECT id FROM EntidadCuenta__c WHERE Cuenta__c =: accId) and isActive__c = true]){
                if(!mapaFilial.containsKey(ec.Entidad_Cuenta__c)) mapaFilial.put(ec.Entidad_Cuenta__c, new List<ContratoFilial__c>());
                List<ContratoFilial__c> tem = mapaFilial.get(ec.Entidad_Cuenta__c);
                tem.add(ec);
                mapaFilial.put(ec.Entidad_Cuenta__c, tem);
            }
        }
        for(Opportunity opp : this.newList){
            /* Modificaciones Televia */
            if(!opp.NTT_IsMigratedOpp__c){
                if(opp.StageName != 'Ganada' && opp.StageName != 'Renegociada'){
                    Boolean esVentaNueva = isRecordType('VentaNueva', opp.RecordTypeId);
                    Boolean esCombustible = (opp.Familia__c == 'Combustible');
                    // 25/02/2026 NTTData:
                    // Para oportunidades VentaNueva NO Combustible, el TipoVenta__c se calcula en
                    // OpportunityHelperTrigger.asignarTipoVentaNoCombustibleEnCreacion y
                    // se conserva en todas las etapas; aquí no lo sobreescribimos ni lo limpiamos.
                    if(esVentaNueva && !esCombustible){
                        continue;
                    }
                    // Asegurar que TipoVenta__c sólo se asigne cuando la oportunidad esté en la etapa 'Implementación'
                    // para el resto de escenarios (Migración, BPfleet, Renegociaciones, etc.) y VentaNueva Combustible.
                    if(opp.StageName == 'Implementación'){
                        // Si el helper ya asignó TipoVenta__c (por ejemplo, Combustible), respetamos ese valor y no lo sobrescribimos.
                        if(opp.TipoVenta__c != null){
                            // mantener el valor asignado previamente
                        } else {
                            if(/*isRecordType('VentaNueva', opp.RecordTypeId) ||*/ isRecordType('Migracion', opp.RecordTypeId) || isRecordType('BPfleet', opp.RecordTypeId)){
                                if(opp.Entidad_Cuenta__c != null && (!mapaFilial.containsKey(opp.Entidad_Cuenta__c) || (mapaFilial.containsKey(opp.Entidad_Cuenta__c) && sameOpp(mapaFilial.get(opp.Entidad_Cuenta__c), opp.Id)))){
                                    opp.TipoVenta__c = 'Simple';
                                } else if(opp.Entidad_Cuenta__c != null && mapaFilial.containsKey(opp.Entidad_Cuenta__c) && esUpselling(mapaFilial.get(opp.Entidad_Cuenta__c), opp.Familia__c, opp.Id, opp.CreatedDate)){
                                    opp.TipoVenta__c = 'Upselling';
                                } else if(opp.Entidad_Cuenta__c != null && mapaFilial.containsKey(opp.Entidad_Cuenta__c) && esCruzada(mapaFilial.get(opp.Entidad_Cuenta__c), opp.Familia__c, opp.Id)){
                                    opp.TipoVenta__c = 'Cruzada';
                                } else {
                                    opp.TipoVenta__c = 'Simple'; //RJP-200824
                                }
                            } else if(isRecordType('RT_Renegotiation', opp.RecordTypeId) || isRecordType('RT_Renegotiation_merchants', opp.RecordTypeId) || isRecordType('Renegociacion_Merchant', opp.RecordTypeId)){
                                opp.TipoVenta__c = 'Renegociación';
                            } else if(isRecordType('SingleOrder', opp.RecordTypeId)){
                                opp.TipoVenta__c = 'Pedido Unico';
                            } else if(isRecordType('Incremento', opp.RecordTypeId)){
                                opp.TipoVenta__c = 'Upselling';
                            }
                        }
                    }
                }
            }
        }
    }

    public static Boolean sameOpp(List<ContratoFilial__c> contratoFilialList, String idOpp){
        Boolean sameOpp = false;
        if(contratoFilialList.size() == 1 && contratoFilialList.get(0).Contrato_2__r.PartidaPresupuesto__r.Quote.OpportunityId != null){
            sameOpp = contratoFilialList.get(0).Contrato_2__r.PartidaPresupuesto__r.Quote.OpportunityId == idOpp;
        }
        return sameOpp;
    }
    public static boolean esCruzada(List<ContratoFilial__c> listSub, String familia, String idOp){
        Boolean res = false;
        for(ContratoFilial__c cf : listSub){
            if(cf.Producto__r.Family != familia
                && (cf.Contrato_2__r.PartidaPresupuesto__r.Quote.OpportunityId == null ||
                cf.Contrato_2__r.PartidaPresupuesto__r.Quote.OpportunityId != idOp)
            ){
                res = true;
            }
        }
        return res;
    }

    public void validarActualizacionFechaEC(){
        List<Opportunity> lisFEC = lisFECGlob;
        lisQliGlob = new List<QuoteLineItem>();
        if(lisFECGlob != null && lisFECGlob.size() > 0){
            lisQliGlob = [Select id,Escalonamiento__c ,Quote.OpportunityId,ServiceDate,NTT_IsMigratedQli__c
                          From QuoteLineItem Where Quote.OpportunityId IN :lisFECGlob and Quote.IsSyncing=true];
        }
        if(!lisFEC.isEmpty()) {
            Map<Id, QuoteLineItem> mapaEscalonamientos = new Map<Id, QuoteLineItem>();
            Map<Id, List<QuoteLineItem>> mapaQli = new Map<Id, List<QuoteLineItem>>();
            for(QuoteLineItem qli : lisQliGlob) {
                /*modificaciones Televia*/
                if(!qli.NTT_IsMigratedQli__c){
                    /*modificaciones Televia*/
                    Id opportunityId = qli.Quote.OpportunityId;
                    if(qli.Escalonamiento__c == null) {
                        mapaEscalonamientos.put(opportunityId, qli);
                    }
                    if(!mapaQli.containsKey(opportunityId)) {
                        mapaQli.put(opportunityId, new List<QuoteLineItem>());
                    }
                    mapaQli.get(opportunityId).add(qli);
                }
            }
            if(trigger.isBefore && trigger.isUpdate) {
                for(Opportunity opp : lisFEC) {
                    /* Modificaciones Televia */
                    if(!opp.NTT_IsMigratedOpp__c){
                        if((opp.StageName == 'Perdida'||opp.StageName == 'Ganada') && opp.StageName == oldMap.get(opp.Id).StageName) {
                            opp.addError('No se puede modificar la fecha estimada de compra o el volumen de emision cuando la Oportunidad está cerrada');
                        }else if(mapaQli.containsKey(opp.Id)) {
                            for(QuoteLineItem qli : mapaQli.get(opp.Id)){
                                if(mapaQli.get(opp.Id).size() > 1 && opp.closeDate == qli.ServiceDate && qli.Escalonamiento__c != null) {
                                    opp.addError('La fecha estimada de compra ya pertenece a un escalonamiento: ' + qli.Id);
                                }
                                if(mapaQli.get(opp.Id).size() > 1 && qli.Escalonamiento__c != null && opp.CloseDate > qli.ServiceDate) {
                                    opp.addError('La fecha estimada de compra no puede ser mayor a ' + qli.ServiceDate);
                                }
                            }
                        }
                    }
                }
            }
            if(trigger.isAfter && trigger.isUpdate){
                List<QuoteLineItem> listQLIup = new List<QuoteLineItem>();
                for(QuoteLineItem qli: mapaEscalonamientos.values()) {
                    if(qli.ServiceDate != newMap.get(qli.Quote.OpportunityId).closeDate){
                        qli.ServiceDate = newMap.get(qli.Quote.OpportunityId).closeDate;
                        listQLIup.add(qli);
                    }
                }
                if(!listQLIup.isEmpty()) {
                    //update listQLIup;
                    List<Database.SaveResult> srList = Database.update(listQLIup, false);
                    RecordSaveErrorLogger.logDmlErrors(listQLIup, srList, 'QuoteLineItem Update');
                }
            }
        }

    /**
     * Method : validateOpportunityBeforeUpdate
     * Purpose: This method is used to prevent update opportunity based on RecordType, StageName, Quote and QuoteLineItem, Contract on before update
     */
    public static void validateOpportunityBeforeUpdate(Map<Id, Opportunity> mapNewOpportunity, Map<Id, Opportunity> mapOldOpportunity){
        List<Id> listClientRegOppId = new List<Id>();
        List<Id> listContractOppId = new List<Id>();
        List<Id> listAnalysisOppId = new List<Id>();
        Map<Id, List<ContentDocumentLink>> mapCilentRegOppIdToContentDocLink = new Map<Id, List<ContentDocumentLink>>();
        Map<Id, List<ContentDocumentLink>> mapContractOppIdToContentDocLink = new Map<Id, List<ContentDocumentLink>>();
        Map<Id, List<ContentDocumentLink>> mapAnalysisOppIdToContentDocLink = new Map<Id, List<ContentDocumentLink>>();
        // Televia inicio
        String  idContrato33='';
        // Televia fin
        Set<Id> setNonFormPaymentOppId = new Set<Id>();
        Map<Id, String> mapOppIdToQuoteDocType = new Map<Id, String>();
        Map<String, List<String>> mapDocTypeToRequiredFields = new Map<String, List<String>>();
        for(Opportunity opportunity : mapNewOpportunity.values()){
            /* Modificaciones Televia */
            if(!opportunity.NTT_IsMigratedOpp__c){
                if(opportunity.StageName == 'Alta de cliente' || opportunity.StageName == 'Alta de Merchant' /*|| opportunity.StageName == 'Alta de servicio'*/) {// 6298 --> Alta de servicio (Descomentar si se activa Tarjetas Virtuales)
                    listClientRegOppId.add(opportunity.Id);
                } else if(opportunity.StageName == 'Contrato') {
                    listContractOppId.add(opportunity.Id);
                }
                // Televia inicio
                if(opportunity.Contrato2__c !=  mapOldOpportunity.get(opportunity.Id).Contrato2__c ){
                    if(!String.isBlank(opportunity.Contrato2__c )){
                        idContrato33 =ValidarContratoOpp(opportunity.Id);
                        if(!String.isBlank(idContrato33)){
                            opportunity.Contrato2__c  = idContrato33;
                        }
                    }
                }
                // Televia fin
                if((opportunity.RecordTypeId == RTCreditRenewalRecordTypeId || opportunity.RecordTypeId == RTChangeParametersRecordTypeId)
                && opportunity.StageName == 'Análisis' && mapOldOpportunity.get(opportunity.Id).StageName != 'Análisis') {
                    listAnalysisOppId.add(opportunity.Id);
                }
            }
        }
        if(!listClientRegOppId.isEmpty() || !listContractOppId.isEmpty() || !listAnalysisOppId.isEmpty()) {
            for(ContentDocumentLink contentDocLink : [SELECT Id, ContentDocumentId, ContentDocument.LatestPublishedVersionId, ContentDocument.Title, LinkedEntityId
                                                      FROM ContentDocumentLink
                                                      WHERE LinkedEntityId IN: listClientRegOppId
                                                          OR LinkedEntityId IN: listContractOppId
                                                          OR LinkedEntityId IN:listAnalysisOppId]) {
                Id opportunityId = contentDocLink.LinkedEntityId;
                if(listClientRegOppId.contains(opportunityId)) {
                    if(!mapCilentRegOppIdToContentDocLink.containsKey(opportunityId)) {
                        mapCilentRegOppIdToContentDocLink.put(opportunityId, new List<ContentDocumentLink>());
                    }
                    mapCilentRegOppIdToContentDocLink.get(opportunityId).add(contentDocLink);
                }
                if(listContractOppId.contains(opportunityId)) {
                    if(!mapContractOppIdToContentDocLink.containsKey(opportunityId)) {
                        mapContractOppIdToContentDocLink.put(opportunityId, new List<ContentDocumentLink>());
                    }
                    mapContractOppIdToContentDocLink.get(opportunityId).add(contentDocLink);
                }
                if(listAnalysisOppId.contains(opportunityId)) {
                    if(!mapAnalysisOppIdToContentDocLink.containsKey(opportunityId)) {
                        mapAnalysisOppIdToContentDocLink.put(opportunityId, new List<ContentDocumentLink>());
                    }
                    mapAnalysisOppIdToContentDocLink.get(opportunityId).add(contentDocLink);
                }
            }
            for(Quote quote : [SELECT Id, OpportunityId, (SELECT Id, FormaPago__c, DocumentType__c FROM QuoteLineItems WHERE Escalonamiento__c = null LIMIT 1)
                               FROM Quote
                               WHERE IsSyncing = true
                                   AND (OpportunityId IN: listClientRegOppId
                                   OR OpportunityId IN: listContractOppId
                                   OR OpportunityId IN: listAnalysisOppId)]) {
                for(QuoteLineItem quoteItem : quote.QuoteLineItems) {
                    if(quoteItem.FormaPago__c == null) {
                        setNonFormPaymentOppId.add(quote.OpportunityId);
                    }
                    if(String.isNotBlank(quoteItem.DocumentType__c)) {
                        mapOppIdToQuoteDocType.put(quote.OpportunityId, quoteItem.DocumentType__c.toLowerCase());
                    }
                }
            }
            if(!mapOppIdToQuoteDocType.isEmpty()) {
                mapDocTypeToRequiredFields = fetchDocRequiredFieldsByType(mapOppIdToQuoteDocType.values());
            }
        }
        setOpportunityContacto(mapNewOpportunity.values());
        validateClientRegOpportunities(listClientRegOppId, mapNewOpportunity, mapOldOpportunity, mapOppIdToQuoteDocType, mapDocTypeToRequiredFields, mapCilentRegOppIdToContentDocLink);
        validateContractOpportunities(listContractOppId, mapNewOpportunity, mapOldOpportunity, mapOppIdToQuoteDocType, mapDocTypeToRequiredFields, mapContractOppIdToContentDocLink, setNonFormPaymentOppId);
        validateAnalysisOpportunities(listAnalysisOppId, mapNewOpportunity, mapOldOpportunity, mapOppIdToQuoteDocType, mapDocTypeToRequiredFields, mapAnalysisOppIdToContentDocLink);
        validateOpportunitiesByContractAddress(mapNewOpportunity.values(), mapOldOpportunity);
        validateOpportunitiesBasedOnContentDocuments(mapOldOpportunity, mapNewOpportunity.values(), mapCilentRegOppIdToContentDocLink);
    }

    private static void validateOpportunitiesByContractAddress(List<Opportunity> listNewOpportunity, Map<Id, Opportunity> mapOldOpportunity) {
        List<Opportunity> listFilteredOpp = new List<Opportunity>();
        Map<Id, Contrato2__c> mapOppIdToContract = new Map<Id, Contrato2__c>();
        Map<Id, Contrato_Direccion__c> mapOppIdToContractDirection = new Map<Id, Contrato_Direccion__c>();
        for(Opportunity opportunity : listNewOpportunity) {
            /* Modificaciones Televia */
            if(!opportunity.NTT_IsMigratedOpp__c){
                if(opportunity.StageName == 'Alta de cliente' || opportunity.StageName == 'Alta de Merchant'){
                    listFilteredOpp.add(opportunity);
                }
            }
        }
        if(listFilteredOpp.isEmpty()) {
            return;
        }
        for(Contrato2__c contract : [SELECT Id, EtapaContrato__c, DireccionesAuxiliares__c, PartidaPresupuesto__r.Quote.OpportunityId,
                                         PartidaPresupuesto__r.meanType__c, PartidaPresupuesto__r.Product2.ProductCode
                                     FROM Contrato2__c
                                     WHERE PartidaPresupuesto__r.Quote.OpportunityId = :listFilteredOpp
                                      // Televia Inicio
                                         AND PartidaPresupuesto__r.Quote.IsSyncing = true and PartidaPresupuesto__r.Product2.ProductCode !=: productCodeTelevia /*'133'*/
                                      // Televia fin
                                     ORDER BY CreatedDate ASC]) {
            mapOppIdToContract.put(contract.PartidaPresupuesto__r.Quote.OpportunityId, contract);
        }
        for(Contrato_Direccion__c contractDirection : [SELECT Id, ContratoId__r.PartidaPresupuesto__r.Quote.OpportunityId
                                                       FROM Contrato_Direccion__c
                                                       WHERE ContratoId__r.PartidaPresupuesto__r.Quote.OpportunityId =: listFilteredOpp]) {
            mapOppIdToContractDirection.put(contractDirection.ContratoId__r.PartidaPresupuesto__r.Quote.OpportunityId, contractDirection);
        }
        for(Opportunity opportunity : listFilteredOpp) {
            /* Modificaciones Televia */
            if(!opportunity.NTT_IsMigratedOpp__c){
                final Id OPP_RECORD_TYPE_ID = opportunity.RecordTypeId;
                Boolean isApplicableRecordType = opportunity.recordTypeId != RTRenegotiationRecordTypeId && opportunity.recordTypeId != newRenegotiationMerchants;
                Boolean isApplicableContractStage = mapOppIdToContract.containsKey(opportunity.Id) && mapOppIdToContract.get(opportunity.Id).EtapaContrato__c != 'Finalizado';
                QuoteLineItem quoteItem = mapOppIdToContract.containsKey(opportunity.Id) ? mapOppIdToContract.get(opportunity.Id).PartidaPresupuesto__r : null;
                Boolean isAppilcableOppStage =
                    (opportunity.StageName == 'Alta de cliente' && mapOldOpportunity.get(opportunity.Id).StageName != 'Alta de cliente')
                    || (opportunity.StageName == 'Alta de Merchant' && mapOldOpportunity.get(opportunity.Id).StageName != 'Alta de Merchant');
                if(isAppilcableOppStage && isApplicableRecordType && isApplicableContractStage){
                    if(!opportunity.BypassValidationRule__c) {
                        if ( OPP_RECORD_TYPE_ID != RenegotiationMerchantFleet ) {
                            opportunity.addError('Necesita Finalizar el contrato para poder pasar a ADV');
                        }
                    }
                }
                Boolean isApplicableContractProduct = quoteItem != null
                    && quoteItem.Product2.ProductCode != '43' && (quoteItem.Product2.ProductCode != '33'  || (quoteItem.Product2.ProductCode == '33' && quoteItem.meanType__c != '' && quoteItem.meanType__c != 'Tarjetas Virtuales'));
                if(!mapOppIdToContractDirection.containsKey(opportunity.Id) && isApplicableRecordType && !affiliateRecordTypes.contains(opportunity.recordTypeId) && mapOppIdToContract.containsKey(opportunity.Id) && mapOppIdToContract.get(opportunity.Id).DireccionesAuxiliares__c == false && isApplicableContractProduct) {
                    if ( OPP_RECORD_TYPE_ID != RenegotiationMerchantFleet ) {
                        opportunity.addError('Necesita agregar una Dirección de Entrega para poder pasar a ADV');
                    }
                }
            }
        }
    }

         */
        public static void oppPerdidaOTCancelada(List<Opportunity> opps, Map<Id, Opportunity> oldMap){
            //Inicialización de un set de Id para almacenar las oportunidades que cambiaron de etapa Perdidas
            Set<Id> oppPerdidads = new Set<Id>();
            if(opps.size() > 0){
                //Loop para verificar de la lista de Opp entrante cuales de ellas cambiaron de etapa y asignarlas al set de Ids
                for(Opportunity opp : opps ) {
                    /* Modificaciones Televia */
                    if(!opp.NTT_IsMigratedOpp__c){
                        // Verificar que la oportunidad está cambiando a StageName = 'Implementación'
                        if(opp.isClosed && opp.StageName == 'Perdida' ) {
                            oppPerdidads.add(opp.Id);
                        }
                    }
                }
                if(oppPerdidads.isEmpty()) {
                    return;
                }
                //Lista que contiene lar ordenes de trabajo relacionadas a las opp entrantes
                List<WorkOrder> ordenesOpp = [Select Id ,Oportunidad__c,Estatus__c,Motivo_del_estatus__c
                                              From WorkOrder
                                              Where Oportunidad__c in : oppPerdidads
                                              AND Estatus__c != 'Concluida'
                                              AND Estatus__c != 'Cancelada'];
                //Se asigna el estatus Cancelada a la WO
                if(!ordenesOpp.isEmpty()) {
                    for(WorkOrder wO : ordenesOpp){
                        wO.Estatus__c = 'Cancelada';
                        wO.Motivo_del_estatus__c = 'Otros';
                    }
                    List<Database.SaveResult> srList = Database.update(ordenesOpp, false);
                    RecordSaveErrorLogger.logDmlErrors(ordenesOpp, srList, 'WorkOrder Update');
                }
            }
        }

        /**
         * ──────────────────────────────────────────────────────────────────────────────────────────────────────────────────┐
         * @Created  30/03/2020  Description: If the opp needs an owner change, validate if the new owner is from the correct
        business unit based on the opp solution.
             * ──────────────────────────────────────────────────────────────────────────────────────────────────────────────────┘
         */
        public static void ownerBUValidation(List<Opportunity> newList, Map<Id, Opportunity> oldMap){
            List<Opportunity> oppsToValidate = new List<Opportunity>();
            Set<String> oppSolutions = new Set<String>();
            Set<String> ownersIds = new Set<String>();
            for(Opportunity opp: newList){
                /* Modificaciones Televia */
                if(!opp.NTT_IsMigratedOpp__c){
                    if(opp.OwnerChangeRequired__c  == true && opp.Familia__c != null && (opp.OwnerId != oldMap.get(opp.Id).OwnerId) && (opp.StageName == oldMap.get(opp.Id).StageName)){
                        oppsToValidate.add(opp);
                        oppSolutions.add(opp.Familia__c);
                        ownersIds.add(opp.OwnerId);
                    }
                }
            }
            if (!oppsToValidate.isEmpty()) {
                Map<String, List<String>> solutionAndRoles = new Map<String, List<String>>();
                Map<Id, User> ownersRoles = new Map<ID, User>([SELECT Id, UserRole.Name FROM User WHERE Id IN: ownersIds]);
                Set<String> solutionSet = new Set<String>();
                for(String solution: oppSolutions){
                    if(solution == 'Ayuda Social'){ solution = 'Ayuda social';
                    }
                    solutionSet.add(solution);
                }
                List<Roles_BusinessUnits_Solutions__mdt> bUSolutions = [SELECT MasterLabel, AllowedSolutions__c
                                                                    FROM Roles_BusinessUnits_Solutions__mdt
                                                                    WHERE AllowedSolutions__c IN : solutionSet];
                for(String solution: oppSolutions){
                    if(solution == 'Ayuda Social'){ solution = 'Ayuda social';
                    }
                    if(!bUSolutions.isEmpty()){
                        for(Roles_BusinessUnits_Solutions__mdt bUConfig: bUSolutions){
                            if(solutionAndRoles.containsKey(solution)){
                                //If the allSolutionsWhithContracts contains the key(acc-allAllowedSolutions) that contains the comtrato filial solution, only add the contrato to the existing value list.
                                List<String> roles = solutionAndRoles.get(solution);
                                roles.add(bUConfig.MasterLabel);
                                solutionAndRoles.put(solution, roles);
                            } else {
                                //If the allSolutionsWhithContracts does not contain the key(acc-allAllowedSolutions) that contains the comtrato filial solution, add the key and initialize the value list with the contrato.
                                solutionAndRoles.put(solution, new List<String> { bUConfig.MasterLabel });
                            }
                        }
                    }
                }
                for (Opportunity opp: oppsToValidate) {
                    String solution = opp.Familia__c;
                    if(solution == 'Ayuda Social'){ solution = 'Ayuda social';
                    }
                    if (solutionAndRoles.containsKey(solution) && ownersRoles.containsKey(opp.OwnerId) && !solutionAndRoles.get(solution).contains(ownersRoles.get(opp.OwnerId).UserRole.Name)) {
                        opp.addError('El ejecutivo asignado no corresponde a la business unit relacionada a la solución asociada a la oportunidad.');
                    } else {
                        opp.OwnerChangeRequired__c = false;
                    }
                }
            }
        }

                private static Set<Id> filterOpportunityIdsForWorkOrderGeneration(Map<Id, Opportunity> mapOldOpportunity, Map<Id, Opportunity> mapNewOpportunity) {
                    Set<Id> setFilteredOpportunitiesId = new Set<Id>();
                    Set<Id> setOppIdToValidate = new Set<Id>();
                    for(Opportunity newOpportunity : mapNewOpportunity.values()) {
                        Opportunity oldOpportunity = mapOldOpportunity.get(newOpportunity.Id);
                        if(newOpportunity.StageName != oldOpportunity.StageName && newOpportunity.StageName == 'Implementación') {
                            setFilteredOpportunitiesId.add(newOpportunity.Id);
                        }
                    }
                    if(setFilteredOpportunitiesId.isEmpty()) {
                        return setOppIdToValidate;
                    }
                    for(Opportunity opportunity : [SELECT Id, (SELECT Id FROM OrdenTrabajo__r LIMIT 1)
                                       FROM Opportunity
                                       WHERE Id IN :setFilteredOpportunitiesId]) {
                        if(opportunity.OrdenTrabajo__r.isEmpty()) {
                            setOppIdToValidate.add(opportunity.Id);
                        }
                    }
                    return setOppIdToValidate;
                }
                private static Map<Id, Id> buildMapOfOpportunityIdToProductCodeFileId(Set<Id> setOpportunityId) {
                    Map<Id, String> mapOppIdToCardName = new Map<Id, String>();
                    Map<String, String> mapProductCodeToCardName = new Map<String, String>();
                    Map<Id, Id> mapOppIdToProductCodeFileId = new Map<Id, Id>();
                    if(setOpportunityId.isEmpty()) {
                        return mapOppIdToProductCodeFileId;
                    }
                    for(PersonalizacionTarjetas__mdt personalCard : [SELECT Id, Label, CodigoProductos__c
                                                         FROM PersonalizacionTarjetas__mdt
                                                         WHERE CodigoProductos__c != null]) {
                        for(String productCode : personalCard.CodigoProductos__c.split(',')) {
                            productCode = productCode.trim();
                            if(String.isNotBlank(productCode)) {
                                mapProductCodeToCardName.put(productCode, personalCard.Label);
                            }
                        }
                    }
                    for(Opportunity opportunity : [SELECT Id, Codigo_Producto_cotizacion__c
                                       FROM Opportunity
                                       WHERE Id IN :setOpportunityId]) {
                        final String PRODUCT_CODE = opportunity.Codigo_Producto_cotizacion__c;
                        if(mapProductCodeToCardName.containsKey(PRODUCT_CODE)) {
                            mapOppIdToCardName.put(opportunity.Id, mapProductCodeToCardName.get(PRODUCT_CODE));
                        }
                    }
                    // Nota: Personalización de tarjetas aún está en Attachment 9/08/2018 @calvarez
                    for(Attachment attachment : [SELECT Id, ParentId, Name
                                     FROM Attachment
                                     WHERE ParentId IN :mapOppIdToCardName.keySet()]) {
                        final Id OPPORTUNITY_ID = attachment.ParentId;
                        if(attachment.Name.toLowerCase().contains(mapOppIdToCardName.get(OPPORTUNITY_ID).toLowerCase())) {
                            mapOppIdToProductCodeFileId.put(OPPORTUNITY_ID, attachment.Id);
                        }
                    }
                    return mapOppIdToProductCodeFileId;
                }

                    public ServiceGroupInfo(ServiceGroups__c serviceGroup, List<ServiceGroupsUser__c> listServiceGroupUser) {
                        name = serviceGroup.Name.toLowerCase();

                    private void populateServiceGroupUserInfo(List<ServiceGroupsUser__c> listServiceGroupUser) {
                        mapMemberKeyToUser = new Map<String, User>();
                        for(ServiceGroupsUser__c serviceGroupUser : listServiceGroupUser) {
                            String[] keyParts = new String[] {};
                            if(IS_ASSIGNMENT_BY_FAMILY) {
                                keyParts.add(serviceGroupUser.Family__c);
                            }
                            keyParts.add(serviceGroupUser.User__c);
                            String memberKey = String.join(keyParts, MEMBER_KEY_SEPARATOR);
                            mapMemberKeyToUser.put(memberKey, serviceGroupUser.User__r);
                        }
                    }
                    public User fetchApplicableUser(String userFamily) {
                        User applicableUser;
                        Integer userIndex;
                        if(String.isBlank(serviceGroup.Users__c)) {   return applicableUser;
                        }
                        List<String> listUserId = serviceGroup.Users__c.split(';');
                        for(userIndex = 0; userIndex < listUserId.size(); userIndex++) {
                            String userId = listUserId[userIndex];
                            String[] keyParts = new String[] {};
                            if(IS_ASSIGNMENT_BY_FAMILY) {
                                keyParts.add(userFamily);
                            }
                            keyParts.add(userId);
                            String memberKey = String.join(keyParts, MEMBER_KEY_SEPARATOR);
                            if(mapMemberKeyToUser.containsKey(memberKey)) {
                                applicableUser = mapMemberKeyToUser.get(memberKey);
                                break;
                            }
                        }

                            listUserId.add(userIdToAdjust);
                            serviceGroup.Users__c = String.join(listUserId, ';');
                            isUserOrderModified = true;
                        }
                        return applicableUser;
                    }
