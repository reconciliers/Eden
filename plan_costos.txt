                    if(discount >= 0 && discount < 5){
                            //Ejecutivo
                        approvalLevel = 1;
                    }else if(discount >= 5 && discount < 10){
                            //Gestor
                        approvalLevel = 2;
                    }else if(discount >= 10 && discount < 20){

                        approvalLevel = 3;
                    }else if(discount >= 20 && discount <= 100){

                        }
