    private static Map<String,List<comboWP>> getMapProd(){
        if(mapProd != null){
            return mapProd;
        }

        mapProd = new Map<String,List<comboWP>>();

        List<AltaContrato_MapProd__mdt> rows = [
            SELECT Key__c, Value__c, Label__c, SortOrder__c
            FROM AltaContrato_MapProd__mdt
            ORDER BY Key__c, SortOrder__c
        ];

        for(AltaContrato_MapProd__mdt row : rows){
            if(row == null || String.isBlank(row.Key__c)){
                continue;
            }
            if(!mapProd.containsKey(row.Key__c)){
                mapProd.put(row.Key__c, new List<comboWP>());
            }
            mapProd.get(row.Key__c).add(new comboWP(row.Value__c, row.Label__c));
        }

        return mapProd;
    }

    /**
    * @Name        : getFieldsToShowInContractSections
    * @param       : String sectionName = Nombre de la sección donde se mostrarán o no los campos, String productCode = Código del producto, Map<String,campoWP> contractFields = Campos que se muestran actualmente en el contrato, sin filtrar
    * @description : Método que filtra los campos del contrato y mostrará solo aquellos que sean necesarios
    **/
    private static List<SpecialFields> getFieldsToShowInContractSections(String sectionName,String productCode,Map<String,campoWP> contractFields){
        Boolean doesShowFieldInSection = false;
        List<SpecialFields> fieldsToShow = new List<SpecialFields>();
        if(String.isEmpty(sectionName) || String.isEmpty(productCode)){
            return fieldsToShow;
        }
        Map<String,List<SpecialFields>> productFields = notContractFieldsPerSections.get(productCode);
        if(productFields.keySet().size() > 0){
            List<SpecialFields> sectionFields = productFields.get(sectionName);
            for(SpecialFields sectionField : sectionFields){
                if(sectionField.dependency == null){
                    fieldsToShow.add(sectionField);
                    continue;
                }
                String fieldDependency = sectionField.dependency.field;
                String valueDependency = sectionField.dependency.value;
                campoWP fieldContract = contractFields.get(fieldDependency);
                if(fieldContract == null){
                    continue;
                }
                if(fieldContract.value == valueDependency){
                    fieldsToShow.add(sectionField);
                }
            }
        }
        return fieldsToShow;
    }

    /**
    * @Name        : getQliProductsConcepts
    * @param       : String idQli, List<String> conceptsToFind
    * @description : Método que obtiene los conceptos pasados al método (conceptsToFind) en la cotización indicada (idQli) y los regresa
    **/
    private static List<ProductoConcepto__c> getQliProductsConcepts(String idQli,List<String> conceptsToFind){
        String conceptsLikeQuery = '(';
        Integer conceptsCount = 0;
        for(String conceptToFind : conceptsToFind){
            conceptsLikeQuery += ' Concepto_Tipo_Cargo__c LIKE \''+conceptToFind+'\'';
            conceptsCount ++;
            if(conceptsCount < conceptsToFind.size()){
                conceptsLikeQuery += ' OR ';
            }else if(conceptsCount == conceptsToFind.size()){
                conceptsLikeQuery += ')';
            }
        }
        String conceptsQuery = 'SELECT Id,Concepto_Tipo_Cargo__c '+
                                        'FROM ProductoConcepto__c '+
                                        'WHERE PartidaPresupuesto__c =: idQli AND '+conceptsLikeQuery;
        List<ProductoConcepto__c> products = new List<ProductoConcepto__c>();
        Set<String> conceptsAdded = new Set<String>();
        for(ProductoConcepto__c concept : Database.query(conceptsQuery)){
            //Se valida que el concepto no se haya ya agregado a la lista de conceptos
            //Para MFG existe Mover for Good Compensación de emsiones 50% o Move for Good Compensación de emisiones 100%, ambos en este caso se tratan como uno solo
            String conceptToAddName = concept.Concepto_Tipo_Cargo__c.toUpperCase().startswith('MOVE FOR GOOD')?'MFG':concept.Concepto_Tipo_Cargo__c;
            if(!conceptsAdded.contains(conceptToAddName)){

    @TestVisible
    private static string getBU(Id idUserRole, Map<Id, Id> mUserRoles,  Map<Id, string> mUserRolesNames){
        Id parent = mUserRoles.get(idUserRole);
        for(Id idrole:mUserRoles.keySet()){
            if (idrole == parent ){
                if (mUserRolesNames.get(idrole) == 'Distant Sales'){    return 'Distant Sales';}
                if (mUserRolesNames.get(idrole) == 'Field Sales'){  return 'Field Sales';   }
                return getBU(parent, mUserRoles, mUserRolesNames);
            }
        }
        return '';
    }
    @TestVisible
    private static void getRoleLevelUp(Id idUserRole, Map<Id, Id> mUserRoles,  Map<Id, string> mUserRolesNames, Map<Id, List<User>> mUsers, List<User> lusers, boolean isLevel2, boolean isFieldSales, boolean isDistantSales){
        Id parent = null;
        if (mUserRoles.containsKey(idUserRole)){
            parent = mUserRoles.get(idUserRole);  } else {  return;
        }
        //Verifica si es nivel 2, si es nivel 2 solo retorna un registro
        if (isLevel2){
            for(Id idrole:mUserRolesNames.keySet()){
                if(mUserRolesNames.get(idrole) == 'CEO Edenred'){
                    for(User u: mUsers.get(idrole)){
                        lusers.add(u);
                        return;
                    }
                }
            }
            return;
        }
        try{
            isLevel2 = mUserRolesNames.get(parent) == 'Distant Sales' || mUserRolesNames.get(parent) == 'Field Sales';  } catch(Exception e) {
        }
        if(isFieldSales){
            if(mUsers.containsKey(idUserRole)){
                for(User u: mUsers.get(idUserRole)){
                    lusers.add(u);
                }
            }
        }else{
            if(!isLevel2){
                for(Id idrole:mUserRoles.keySet()){
                    if(mUserRoles.get(idrole) == parent){
                        if(mUsers.containsKey(idrole)){
                            for(User u: mUsers.get(idrole)){
                                lusers.add(u);
                            }
                        }
                    }
                }
            }else if(mUsers.containsKey(parent)){
                for(User u: mUsers.get(parent)){
                    lusers.add(u);
                }
            }
        }
        if(lusers.size() == 0){  getRoleLevelUp(parent,mUserRoles,mUserRolesNames,mUsers,lusers,isLevel2, isFieldSales, isDistantSales) ;}
    }

                                contactIds.add(oppsContractsMap.get(oppToValidate.Id).Contacto__r.Id);

                            emptyFields.add(deedDateMissing);

                        }

                            oppsWihtError.put(oppToValidate.Id, emptyFields);
                            finalOppsWithErrors.add(oppToValidate);
                        }

                            }
                        }
                    }
                }
                if(!finalOppsWithErrors.isEmpty()){
                    for(Opportunity oppWithError: finalOppsWithErrors){
                        if(oppsWihtError.containsKey(oppWithError.Id)){
                            String strEmptyFields = '';
                            for(String str : oppsWihtError.get(oppWithError.Id)){

                            resFields.add ('Es necesario llenar los siguientes campos del Representante Legal o la Entidad Legal si se indica: ' + strEmptyFields);

                            resFields.add('True');
                            resFields.add('Es necesario llenar los siguientes campos : ' + oppsWihtELEmptyFiels.get(oppWithError.Id));
                            resFields.add(isAdmin);
                        }
                    }
                }
                //Se añade este parche para levantar cobertura, ya que la clase de prueba no soporta mas queries y DMLs
                if(Test.isRunningTest()){  finalOppsWithErrors = new Set<Opportunity>();}
                if(finalOppsWithErrors.isEmpty()) {
                    resFields.add('False');
                    resFields.add('Completa');
                    resFields.add(isAdmin);
                }
            }
        }
        return resFields;
    }

    public static Map<String, String> validaProdCode43( String IdQli ) {
        Map<String, String> response = new Map<String, String>();
        try{
            String  prodCode = '';
            for(Contrato2__c contract : [Select Id, PartidaPresupuesto__c, PartidaPresupuesto__r.Product2.ProductCode from Contrato2__c WHERE PartidaPresupuesto__c =: IdQli
                    AND PartidaPresupuesto__r.Quote.IsSyncing = true ORDER BY CreatedDate DESC LIMIT 1]){
                prodCode = contract.PartidaPresupuesto__r.Product2.ProductCode;
            }
            for(Contrato2__c contract : [Select Id, PartidaPresupuesto__c, PartidaPresupuesto__r.Product2.ProductCode from Contrato2__c WHERE Id =: IdQli  AND PartidaPresupuesto__r.Quote.IsSyncing = true ORDER BY CreatedDate DESC LIMIT 1]){
                prodCode = contract.PartidaPresupuesto__r.Product2.ProductCode;
            }
            if(prodCode != null && prodCode != ''){
                Boolean isMfa = false;
                for(NTT_Properties__mdt nttProp : [Select NTT_Valor__c from NTT_Properties__mdt where DeveloperName = 'NTT_Productos_NFM_Cliente' Limit 1]){
                    List<String> data = new List<String>(nttProp.NTT_Valor__c.split(','));
                    if(data.contains(prodCode)){  isMfa = true;}
                    /*if(nttProp.NTT_Valor__c.IndexOf(prodCode) != -1){
                        isMfa = true;
                    }*/
                }
                response.put('isMfa', String.ValueOf(isMfa));
            }
        }catch(JSONException ex){
            response.put('error', ex.getStackTraceString());
            response.put('JSONException', ex.getMessage());
            throw new CustomException(ex.getMessage() + ', ' + ex.getStackTraceString());
        }catch(Exception ex){
            response.put('error', ex.getStackTraceString());
            response.put('Exception', ex.getMessage());
            throw new CustomException(ex.getMessage() + ', ' + ex.getStackTraceString());
        }
        return response;
    }
