        if(oppId == null) throw new AuraHandledException('No se ha recibido Id de la oportunidad');
        InitData initData = new InitData();
        List<QuoteLineItem> listQuoteLine = new List<QuoteLineItem>();
        Opportunity opportunity = [
			SELECT Id, Name, Entidad_Cuenta__c, GLB_Cambio_Condiciones_Comerciales__c, GLB_Cambio_Condiciones_Facturacion__c,
				StageName, Contrato2__c, Product__c, Product__r.ProductCode,Codigo_Producto_cotizacion__c,SyncedQuoteId,QuotesInApprovalProcess__c,
				GLB_Contratos_Renegociacion__c, Account.Type, RenegotiationType__c, Account.Status__c, MX_Alta_del_programa_MFG__c,RecordType.DeveloperName
			FROM Opportunity
			WHERE Id =: oppId
		];
        List<Quote> listQuote = [
            SELECT Id, (SELECT Id FROM QuoteLineItems WHERE Quote.IsSyncing = true AND Escalonamiento__c = null)
            FROM Quote
            WHERE OpportunityId =: oppId
        ];

            for(Quote quote : listQuote) {
                if(quote.QuoteLineItems != null && !quote.QuoteLineItems.isEmpty() && quote.QuoteLineItems.size() > 0) {
                    for(QuoteLineItem quoteLine : quote.QuoteLineItems) {
                        if(quoteLine.Id != null) {
                            listQuoteLine.add(quoteLine);
                        }
                    }
                }
            }
        }
        initData.opportunity = opportunity;
        initData.listQuote = listQuote;
        initData.listQuoteLine = listQuoteLine;
		initData.contractsIds = new List<String>();
		//Modificar logica primer if, ya que los contratos filiales no se vincularan al nuevo contrato2
		if(opportunity.GLB_Contratos_Renegociacion__c != null) {
			/*List<ContratoFilial__c> conts = [

			for(ContratoFilial__c contr: conts) {
				initData.contractsIds.add(contr.Contrato_2__c);
			}*/
			initData.contractsIds = opportunity.GLB_Contratos_Renegociacion__c.split(',');
		} else if(opportunity.GLB_Cambio_Condiciones_Comerciales__c && opportunity.GLB_Cambio_Condiciones_Facturacion__c
			&& opportunity.Product__r.ProductCode == '33'

	public static List<String> findContracts(String oppId) {
		QuoteLineItem qli = [
			SELECT Id, OpportunityLineItemId
			FROM QuoteLineItem
             // Televia Inicio
			WHERE OpportunityLineItem.OpportunityId =: oppId and Product2.ProductCode !=:productCodeTelevia /*'133'*/
             // Televia fin
		];
		List<RenegotiationProduct__c> renegotiationProducts = [
			SELECT Name
			FROM RenegotiationProduct__c
			WHERE QuoteProduct__c =: qli.Id
			AND isRenegotiation__c = false
		];
		List<String> contractNames = new List<String>();
		for(RenegotiationProduct__c renegotiationProduct: renegotiationProducts) {
			contractNames.add(renegotiationProduct.Name);
		}
		List<ContratoFilial__c> contratosFiliales = [
			SELECT Id
			FROM ContratoFilial__c
			WHERE Name IN: contractNames
		];
		List<String> idContratos = new List<String>();
		for(ContratoFilial__c contratoFilial: contratosFiliales) {
			idContratos.add(contratoFilial.Id);
		}
		return idContratos;
	}
    public static Opportunity getDataOpportunity(String oppId) {
        Opportunity opportunity = [
			SELECT Id, AccountId, Product__c, BypassRenegotiation__c,RecordType.DeveloperName
			FROM Opportunity
			WHERE Id =: oppId
		] ;
        if(opportunity.AccountId == null) throw new AuraHandledException('La oportunidad no contiene Id de Cuenta');
        if(opportunity.Product__c == null) throw new AuraHandledException('La oportunidad no contiene Id de Producto');
        return opportunity;
    }
