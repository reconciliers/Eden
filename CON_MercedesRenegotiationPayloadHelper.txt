    public static Decimal chargeAmount(ProductoConcepto__c productConcept) {
        if (productConcept == null) {
            return null;
        }
        Decimal listPrice = productConcept.PrecioLista__c;
        Decimal finalPrice = productConcept.PrecioFinal__c;
        Boolean hasDiscount = productConcept.Bonificacion__c != null && productConcept.Bonificacion__c > 0;
        if (!hasDiscount && finalPrice != null && (listPrice == null || finalPrice > listPrice)) {
            return finalPrice;
        }
        return listPrice != null ? listPrice : finalPrice;
    }

    public static Boolean applyAnnualChargeFrequency(
        NTTER_ClientToMercedesWrapper.Charges charge,
        Integer billingFrequencyIdentifier,
        String billingRange,
        DateTime annualReferenceDate
    ) {
        if (charge == null || charge.identifier != ANNUAL_FEE_CHARGE_IDENTIFIER) {
            return false;
        }

        String currentApplicationDay = charge.frecuency == null
            ? null
            : charge.frecuency.application_day;
        NTTER_ClientToMercedesWrapper.Frecuency annualFrequency =
            new NTTER_ClientToMercedesWrapper.Frecuency();
        annualFrequency.identifier = billingFrequencyIdentifier;
        annualFrequency.range = billingRange;
        annualFrequency.application_id = ANNUAL_APPLICATION_FREQUENCY_ID;
        annualFrequency.application_day = isValidAnnualApplicationDay(currentApplicationDay)
            ? currentApplicationDay
            : formatAnnualApplicationDay(annualReferenceDate);
        charge.frecuency = annualFrequency;
        return true;
    }

    private static Boolean isValidAnnualApplicationDay(String value) {
        return !String.isBlank(value) &&
            Pattern.matches('^(L|[1-9]|1[0-9]|2[0-8]),([1-9]|1[0-2])$', value);
    }

    private static String formatAnnualApplicationDay(DateTime referenceDate) {
        if (referenceDate == null) {
            return null;
        }
        Date annualDate = referenceDate.dateGmt();
        String dayValue = annualDate.day() > 28
            ? 'L'
            : String.valueOf(annualDate.day());
        return dayValue + ',' + String.valueOf(annualDate.month());
    }

    public static String removeAccents(String text) {
        if (String.isBlank(text)) {
            return text;
        }
        for (String key : accents.keySet()) {
            text = text.replaceAll(key, accents.get(key));
        }
        return text;
    }

    public static List<NTTER_ClientToMercedesWrapper.Accounts> buildRenegotiationAccountReferences(
        List<NTTER_ClientToMercedesWrapper.Accounts> sourceAccounts,
        String conditionRef,
        String productCode
    ) {
        List<NTTER_ClientToMercedesWrapper.Accounts> references = new List<NTTER_ClientToMercedesWrapper.Accounts>();
        if (sourceAccounts == null || String.isBlank(conditionRef)) {
            return references;
        }
        Integer fallbackProductId = parseMercedesProductCode(productCode);
        for (NTTER_ClientToMercedesWrapper.Accounts sourceAccount : sourceAccounts) {
            if (sourceAccount == null || String.isBlank(sourceAccount.identifier)) {
                continue;
            }
            NTTER_ClientToMercedesWrapper.Accounts accountRef = new NTTER_ClientToMercedesWrapper.Accounts();
            accountRef.identifier = sourceAccount.identifier;
            accountRef.products = new List<NTTER_ClientToMercedesWrapper.Products>();
            if (sourceAccount.products != null) {
                for (NTTER_ClientToMercedesWrapper.Products sourceProduct : sourceAccount.products) {
                    Integer productId = sourceProduct == null ? null : sourceProduct.identifier;
                    if (productId == null) {
                        productId = fallbackProductId;
                    }
                    if (productId == null) {
                        continue;
                    }
                    NTTER_ClientToMercedesWrapper.Products productRef = new NTTER_ClientToMercedesWrapper.Products();
                    productRef.identifier = productId;
                    productRef.condition_ref = conditionRef;
                    accountRef.products.add(productRef);
                }
            }
            if (accountRef.products.isEmpty() && fallbackProductId != null) {
                NTTER_ClientToMercedesWrapper.Products productRef = new NTTER_ClientToMercedesWrapper.Products();
                productRef.identifier = fallbackProductId;
                productRef.condition_ref = conditionRef;
                accountRef.products.add(productRef);
            }
            if (!accountRef.products.isEmpty()) {
                references.add(accountRef);
            }
        }
        return references;
    }

    public static Boolean hasWrapperAccountProductReference(
        List<NTTER_ClientToMercedesWrapper.Accounts> accounts
    ) {
        if (accounts == null || accounts.isEmpty()) {
            return false;
        }
        Boolean hasProductReference = false;
        for (NTTER_ClientToMercedesWrapper.Accounts account : accounts) {
            if (account == null || String.isBlank(account.identifier) || account.products == null || account.products.isEmpty()) {
                return false;
            }
            for (NTTER_ClientToMercedesWrapper.Products product : account.products) {
                if (
                    product != null &&
                    product.identifier != null &&
                    String.isNotBlank(product.condition_ref)
                ) {
                    hasProductReference = true;
                }
            }
        }
        return hasProductReference;
    }

    public static Boolean hasActionableWrapperAccountProduct(
        List<NTTER_ClientToMercedesWrapper.Accounts> accounts
    ) {
        if (accounts == null || accounts.isEmpty()) {
            return false;
        }
        Boolean hasActionableProduct = false;
        for (NTTER_ClientToMercedesWrapper.Accounts account : accounts) {
            if (account == null || account.action_type == null || account.products == null || account.products.isEmpty()) {
                return false;
            }
            for (NTTER_ClientToMercedesWrapper.Products product : account.products) {
                if (
                    product != null &&
                    product.action_type != null &&
                    String.isNotBlank(product.condition_ref) &&
                    product.items != null &&
                    !product.items.isEmpty()
                ) {
                    hasActionableProduct = true;
                }
            }
        }
        return hasActionableProduct;
    }

    private static Integer parseMercedesProductCode(String productCode) {
        if (String.isBlank(productCode)) {
            return null;
        }
        try {
            return Integer.valueOf(productCode);
        } catch (Exception ex) {
            return null;
        }
    }

    private static String normalizeCatalogKey(String value) {
        String normalized = removeAccents(value);
        return String.isBlank(normalized) ? null : normalized.toUpperCase();
    }

    private static Boolean hasValidMercedesUpdateAccounts(Object accountsRaw) {
        Boolean hasValidProduct = false;
        for (Object accountRaw : asObjectList(accountsRaw)) {
            Map<String, Object> account = asObjectMap(accountRaw);
            if (!isValidMercedesUpdateAccount(account)) {
                return false;
            }
            Boolean accountHasAction = hasNonBlankValue(account.get('action_type'));
            Boolean accountHasProduct = false;
            for (Object productRaw : asObjectList(account.get('products'))) {
                Map<String, Object> product = asObjectMap(productRaw);
                if (isActionableMercedesUpdateProduct(product)) {
                    if (!accountHasAction) {
                        return false;
                    }
                    accountHasProduct = true;
                    hasValidProduct = true;
                } else if (!accountHasAction && isReferenceMercedesUpdateProduct(product)) {
                    accountHasProduct = true;
                    hasValidProduct = true;
                } else {
                    return false;
                }
            }
            if (!accountHasProduct) {
                return false;
            }
        }
        return hasValidProduct;
    }

    private static Boolean isActionableMercedesUpdateProduct(Map<String, Object> product) {
        Object actionType = product.get('action_type');
        return
            actionType != null &&
            String.isNotBlank(String.valueOf(actionType)) &&
            hasNonBlankValue(product.get('condition_ref')) &&
            !asObjectList(product.get('items')).isEmpty();
    }

    private static Boolean isReferenceMercedesUpdateProduct(Map<String, Object> product) {
        return
            product.get('identifier') != null &&
            hasNonBlankValue(product.get('condition_ref')) &&
            product.get('action_type') == null;
    }

    private static Boolean isValidMercedesUpdateAccount(Map<String, Object> account) {
        if (account.isEmpty()) {
            return false;
        }
        if (!isCanonicalMercedesAccountIdentifier(account.get('identifier'))) {
            return false;
        }
        for (String fieldName : new List<String>{
            'account_type_id',
            'brand_id',
            'business_name',
            'commercial_name',
            'country_id',
            'credit_limit',
            'economic',
            'fiscal',
            'is_fiscal_key',
            'latitude',
            'longitude',
            'segment_type_id',
            'tax_zone_id',
            'utc_zone_id'
        }) {
            if (account.containsKey(fieldName) && isBlankMercedesAccountValue(account.get(fieldName))) {
                return false;
            }
        }
        return true;
    }

    private static Boolean isCanonicalMercedesAccountIdentifier(Object rawValue) {
        if (rawValue == null || String.isBlank(String.valueOf(rawValue))) {
            return false;
        }
        return Pattern.matches('^\\d{3}-\\d{7}-\\d{5}$', String.valueOf(rawValue));
    }

    private static Boolean isBlankMercedesAccountValue(Object rawValue) {
        return rawValue == null || (rawValue instanceof String && String.isBlank((String) rawValue));
    }

    private static Boolean hasNonBlankValue(Object rawValue) {
        return rawValue != null && String.isNotBlank(String.valueOf(rawValue));
    }

    private static List<Object> asObjectList(Object rawValue) {
        if (rawValue instanceof List<Object>) {
            return (List<Object>) rawValue;
        }
        return new List<Object>();
    }

    private static Map<String, Object> asObjectMap(Object rawValue) {
        if (rawValue instanceof Map<String, Object>) {
            return (Map<String, Object>) rawValue;
        }
        return new Map<String, Object>();
    }
