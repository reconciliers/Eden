	@TestVisible
	private static List<NTT_MercedesSnapshotContext> buildConfirmedMigrationSnapshotContexts(
		Map<Id, Contrato2__c> mapContratos,
		Boolean success,
		String requestBody,
		Object responseBody
	) {
		List<NTT_MercedesSnapshotContext> contexts = new List<NTT_MercedesSnapshotContext>();
		if (success != true || String.isBlank(requestBody) || mapContratos == null || mapContratos.isEmpty()) {
			return contexts;
		}

		Set<String> processedSnapshotKeys = new Set<String>();
		String responseBodyText = responseBody == null ? null : String.valueOf(responseBody);

		for (Contrato2__c contrato : mapContratos.values()) {
			String codigoAS400 = resolveMercedesSnapshotCodigoAS400(contrato);
			String productCode = resolveMercedesSnapshotProductCode(contrato);
			if (contrato == null || contrato.Id == null || String.isBlank(codigoAS400) || String.isBlank(productCode)) {
				continue;
			}

			String snapshotKey = codigoAS400 + '|' + productCode;
			if (processedSnapshotKeys.contains(snapshotKey)) {
				continue;
			}
			processedSnapshotKeys.add(snapshotKey);

			contexts.add(NTT_MercedesSnapshotContext.forClientsSuccess(
				NTT_MercedesSnapshotContext.SOURCE_TYPE_MIGRACION,
				String.valueOf(contrato.Id),
				codigoAS400,
				null,
				contrato.Id,
				productCode,
				requestBody,
				responseBodyText
			));
		}

		return contexts;
	}

	private static String resolveMercedesSnapshotCodigoAS400(Contrato2__c contrato) {
		try {
			return contrato != null ? contrato.CodigoAS400__c : null;
		} catch (Exception ex) {
			return null;
		}
	}

	private static String resolveMercedesSnapshotProductCode(Contrato2__c contrato) {
		try {
			return contrato != null &&
				contrato.PartidaPresupuesto__r != null &&
				contrato.PartidaPresupuesto__r.Product2 != null
				? contrato.PartidaPresupuesto__r.Product2.ProductCode
				: null;
		} catch (Exception ex) {
			return null;
		}
	}

	public static String handleDaySelected(Integer chargeFrecuencyId, Contrato2__c contrato) {

			'Lunes' => 'Monday',
			'Martes'=> 'Tuesday',
			'Miércoles'=> 'Wednesday',
			'Jueves'=> 'Thursday',
			'Viernes'=> 'Friday',
			'Sábado'=> 'Saturday',
			'Domingo'=> 'Sunday',
			'Ultimo dia del mes'=> 'L',
			'1 y 16'=> '1,16'
		};
		if(chargeFrecuencyId == 4) { //mensual - respeta el dia seleccionado
			if(daySelected == null && contrato.GLB_Facturacion_dia_Especifico__c != null) {
				daySelected = contrato.GLB_Facturacion_dia_Especifico__c;
			} else {
				daySelected = '1';
			}

			daySelected = dayNumber > 28 ? '28' : daySelected;
		}
		if (contrato.GLB_Facturacion_dia_Especifico__c != null && daySelectedCatalog.get(contrato.GLB_Facturacion_dia_Especifico__c) != null) {
			daySelected = daySelectedCatalog.get(contrato.GLB_Facturacion_dia_Especifico__c);

    public Static String convertDate(DateTime dateInput, Boolean toDateTime, Boolean hasTime, Boolean zeroHour){
		DateTime fechaOriginal;
		Integer OFFSET_SIGN = hasTime ? -1 : 1;
        TimeZone tz = UserInfo.getTimeZone();
        Integer HORAS_OFFSET = tz.getOffset(DateTime.Now()) * 60000 * OFFSET_SIGN;
		if (dateInput != null) {
			fechaOriginal = dateInput;
		} else {
			return null;
		}
        Long TimeHoras = fechaOriginal.getTime() + HORAS_OFFSET;
		fechaOriginal = DateTime.newInstance(TimeHoras);
		Integer year = 0;
        String nuevaFecha = '';
        String mes = '';
        String month = '';
        String dia = '';
        String day = '';
		if (toDateTime) {
			year = fechaOriginal.year();
            mes = String.ValueOf(fechaOriginal.Month() + 1);
			month = '0' + mes;
            dia = String.ValueOf(fechaOriginal.Day());
			day = '0' + dia;
			nuevaFecha = zeroHour ? year + '-' + month + '-' + day + ' 00:00:00Z' : year + '-' + month + '-' + day + ' 23:59:00Z';
			return nuevaFecha;
		} else {
			year = fechaOriginal.year();
			mes = String.ValueOf(fechaOriginal.Month() + 1);
			month = '0' + mes;
			dia = String.ValueOf(fechaOriginal.Day());
			day = '0' + dia;
			nuevaFecha = year + '-' + month + '-' + day;
			return nuevaFecha;
		}
	}

    public static String generateRandomInteger() {
        // Generar un número aleatorio de 10 dígitos
        String randomNumber = '';
        for(Integer i = 0; i < 10; i++) {
            Integer digit = Math.mod(Math.abs(Crypto.getRandomInteger()), 10);
            randomNumber += digit;
        }
        return randomNumber;
    }

	@TestVisible
	private static void applyGruposFacturacion(
		Contrato2__c contrato,
		List<Grupo_de_facturacion__c> grupos,
		List<NTTER_ClientToMercedesWrappBase.BillingEntitie> lstBillingEntities,
		List<NTTER_ClientToMercedesWrappBase.Address> addressesContrato,
		String leyendaFacturacion
	){
		if (contrato == null || grupos == null || grupos.isEmpty() || lstBillingEntities == null) {
			return;
		}

		// Direcciones fiscales de los grupos (para elegir delivery distinto)
		Set<String> groupFiscalAddressIds = new Set<String>();
		for (Grupo_de_facturacion__c gf : grupos) {
			if (gf != null && gf.Entidad_Legal__r != null && String.isNotBlank(gf.Entidad_Legal__r.Address__c)) {
				groupFiscalAddressIds.add(gf.Entidad_Legal__r.Address__c);
			}
		}

		// Buscar una dirección del contrato que NO sea ninguna address fiscal de grupo
		String distinctDeliveryAddressId;
		if (addressesContrato != null) {
			for (NTTER_ClientToMercedesWrappBase.Address a : addressesContrato) {
				if (a != null && String.isNotBlank(a.addressCode) && !groupFiscalAddressIds.contains(a.addressCode)) {
					distinctDeliveryAddressId = a.addressCode;
					break;
				}
			}
		}
		if (String.isBlank(distinctDeliveryAddressId)) {
			distinctDeliveryAddressId = contrato.FiscalAddress__c;
		}

		// Agregar billing entities por grupo (como ADV Container)
		for (Grupo_de_facturacion__c gf : grupos) {
			if (gf == null || String.isBlank(gf.CodigoAS400__c)) continue;

			NTTER_ClientToMercedesWrappBase.BillingEntitie billingEntity = new NTTER_ClientToMercedesWrappBase.BillingEntitie();
			billingEntity.billingEntityCode = contrato.Id + '-' + gf.NTT_GrupoRaizId__c;
			billingEntity.billingEntityName = gf.Name;
			billingEntity.billingBusinessName = gf.Entidad_Legal__r != null ? gf.Entidad_Legal__r.RazonSocial__c : gf.Razon_Social__c;
			billingEntity.bankAccountCode = lstBillingEntities.size() > 0 ? lstBillingEntities[0].bankAccountCode : null;
			billingEntity.deliveryAddressCode = distinctDeliveryAddressId;
			billingEntity.fiscalAddressCode = gf.Entidad_Legal__r != null ? gf.Entidad_Legal__r.Address__c : null;
			billingEntity.billingComplementaryText = String.isNotBlank(gf.Leyenda_de_facturacion__c) ? gf.Leyenda_de_facturacion__c : leyendaFacturacion;
			billingEntity.billingEntityContacts = (lstBillingEntities.size() > 0) ? lstBillingEntities[0].billingEntityContacts : new List<NTTER_ClientToMercedesWrappBase.BillingEntityContact>();
			billingEntity.accountCode = gf.CodigoAS400__c;
			billingEntity.fiscalKeyValue = gf.RFC_Text__c;
			billingEntity.fiscalRegimenId = String.isNotBlank(gf.Regimen_Fiscal__c) ? Integer.valueOf(gf.Regimen_Fiscal__c) : null;
			billingEntity.isDefault = false;

			lstBillingEntities.add(billingEntity);
		}

		// Ajustar addressTypes a 5 cuando la dirección coincide con fiscal address del grupo
		if (addressesContrato != null) {
			for (NTTER_ClientToMercedesWrappBase.Address addr : addressesContrato) {
				if (addr == null || String.isBlank(addr.addressCode)) continue;
				if (groupFiscalAddressIds.contains(addr.addressCode)) {
					addr.addressTypeId = 5;
					addr.addressTypes = new List<NTTER_ClientToMercedesWrappBase.AdressTypes>();
					NTTER_ClientToMercedesWrappBase.AdressTypes t = new NTTER_ClientToMercedesWrappBase.AdressTypes();
					t.addressTypeId = 5;
					addr.addressTypes.add(t);
				}
			}
		}
	}
