    public static String createDocuSingEnvelope(String idOpp){
		dsfs__EnvelopeConfiguration__c lstconfiguracion = [SELECT id from dsfs__EnvelopeConfiguration__c where name = 'Edenred' limit 1];
        dsfs__DocuSign_Envelope__c de=new dsfs__DocuSign_Envelope__c(
            dsfs__DocuSign_Email_Subject__c = 'Documentos pendientes de firma',
            dsfs__DocuSign_Email_Message__c = 'Enviamos esta solicitud de firma electrónica, por favor revise el documento y fírmelo electrónicamente',
            dsfs__EnvelopeConfiguration__c = lstconfiguracion.id,
            dsfs__Source_Object__c = idOpp);
        insert de;
        return de.Id;
    }
	public static String createDocuSignEnvelopeDocument(String idDsE, String idDoc, String nameDoc){
    	dsfs__DocuSign_Envelope_Document__c ded = new dsfs__DocuSign_Envelope_Document__c(
        	dsfs__DocuSign_EnvelopeID__c = idDsE,
            dsfs__Extension__c = 'pdf',
            //dsfs__SFDocument_Type__c = 'Attachment',
            dsfs__External_Document_Id__c = idDoc,
            dsfs__Document_Name__c = nameDoc);
        insert ded;
        return ded.Id;
    }
    public static String createDocuSignEnvelopeRecipient(String idDsE, String emailSign, String nameSign, String idSign, integer order){
        dsfs__DocuSign_Envelope_Recipient__c der = new dsfs__DocuSign_Envelope_Recipient__c(
            dsfs__SignNow__c = false,
            dsfs__SignInPersonName__c = nameSign,
            dsfs__SignInPersonEmail__c = emailSign,
            dsfs__Routing_Order__c = order,
            dsfs__RoleValue__c = order,
            dsfs__RoleName__c = 'Signer ' + order,
            dsfs__Recipient_ID__c = order,
            dsfs__Recipient_Email__c = emailSign,
            dsfs__ID_Check__c = false,
            dsfs__DocuSign_Signer_Type__c = 'Signer',
            dsfs__DocuSign_Signature_Name__c = nameSign,
            dsfs__DocuSign_Recipient_Role__c = 'Customer ' + order,
            dsfs__DocuSign_EnvelopeID__c = idDsE);
        if(order != 1){
            der.dsfs__Salesforce_Recipient_Type__c = 'Contact';
            der.dsfs__DSER_ContactID__c = idSign;
        }
        else{
            der.dsfs__Salesforce_Recipient_Type__c = 'User';
            der.dsfs__DSER_UserID__c = idSign;
        }
		insert der;
        return der.Id;
    }
