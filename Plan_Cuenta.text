                        ctcUtils.getValueFromQuote = function(fieldName){
                            let fieldValue = 0;
                            const QUOTE = ctcUtils.app.cots[0].productos[0];
                            switch(fieldName){
                                case 'Intereses':
                                    //ctcUtils.isNotInterestCharge = QUOTE.GLB_InteresesMoratorios__c == 'No';
                                    const [TIIE,VALUE] = QUOTE.GLB_CobroInteresesMoratorios__c == undefined?[null,null]: QUOTE.GLB_CobroInteresesMoratorios__c.split('+');
                                    let interestValue = 0;
                                    if(ctcUtils.isStringNotEmpty(VALUE)){
                                        //Para el escenario de TIIE + 0%, para que no se malinterprete por el escenario de No aplica o 0%, se manda un 1 a comparar en la matriz
                                        interestValue = parseFloat(VALUE) == 0?1:parseFloat(VALUE);
                                    }else if(ctcUtils.isStringNotEmpty(TIIE)){
                                        //Si está vacío el value pero no el tiie, está en los escenarios de 0 o No aplica
                                        //Si la conversión a float es NaN, se seleccionó no aplica, se pone por default valor 0
                                        interestValue = parseFloat(TIIE)?parseFloat(TIIE):0;
                                    }
                                    fieldValue = QUOTE.GLB_InteresesMoratorios__c != 'No'?interestValue:0;
                                    break; 
                                case 'Monto':
                                    fieldValue = QUOTE.GLB_LineaCreditoPreautorizada__c;
                                    break;
                                case 'DiasPago':
                                    fieldValue = QUOTE.GLB_TerminoDePago__c;
                                    break;
                                default: 
                            }
                            return fieldValue;
                        }

                        /**
                            * Valida las matrices de aprobación obtenidas en la mtd NTT_Matriz_Aprobacion_mdt y en base a ellas, decide que flujo de aprobación se ocupará
                            * 
                            * @param fieldType indica que campo es el que ejecutó dicho método (Término de pago, línea de crédito preautorizada o Cobro de intereses)
                                                este campo se llamará igual que el uno de los registros de la metadata NTT_Matriz_Aprobacion_mdt, es decir, si en la mdt existe el registro DiasPago_33
                                                este campo deberá llamarse DiasPago.
                            * @param fieldValue Valor que se evaluará en los rangos de la matriz coincidente
                            * @returns void
                        */
                        ctcUtils.evaluateTheApprovalStage = function(fieldType,fieldValue){
                            try{
                                const APPROVALS_MATRIX = ctcUtils.app.approvalMatrix;
                                if(APPROVALS_MATRIX != undefined && APPROVALS_MATRIX.length > 0 && ctcUtils.isStringNotEmpty(ctcUtils.app.ProductCode) && ctcUtils.isStringNotEmpty(fieldType) && ctcUtils.isStringNotEmpty(fieldValue)){
                                    //Buscamos la matriz a evaluar
                                    let ownerRole = ctcUtils.ownerRoleToSearch != ''?ctcUtils.ownerRoleToSearch:ctcUtils.app.opportunity.Owner.Role__c.split(' ').join('_');
                                    //ESTA NO let ownerRole = ctcUtils.app.opportunity.Owner.Role__c.split(' ').join('_');
                                    //const APPROVAL_MATRIX_NAME = fieldType+'_'+ownerRole+'_'+ctcUtils.app.ProductCode;
                                    const APPROVAL_MATRIX_NAME = fieldType+'_'+ownerRole+'_GE';
                                    //const APPROVAL_MATRIX_NAME = fieldType+'_Move_GE';
                                    const APPROVAL_MATRIX = APPROVALS_MATRIX.find(e => e.DeveloperName === APPROVAL_MATRIX_NAME);
                                    if(Object.keys(APPROVAL_MATRIX).length > 0){
                                        //Hubo una matriz de aprobación que coincidió
                                        const RANKS_LIST = APPROVAL_MATRIX.NTT_Rangos__c.split(';');
                                        const FLOWS_LIST = APPROVAL_MATRIX.NTT_Flujos__c.split(';'); 
                                        const HIERARCHY_LIST = APPROVAL_MATRIX.NTT_Nivel_de_Jerarquia__c.split(';');
                                        for(let i = 0; i< RANKS_LIST.length; i++){
                                            //Se guarda el monto actual del flujo
                                            ctcUtils.approvalMatrixFlows[ctcUtils.app.ProductCode][fieldType] = fieldValue;
                                            //Se obtiene el valor mínimo y máximo de cada rango y sobre ese evaluamos el valor
                                            const [MIN_VALUE,MAX_VALUE] = RANKS_LIST[i].split(',');
                                            //Se valida fieldValue == MIN_VALUE para los escenarios donde el MIN_VALUE es cero, si MAX_VALUE == undefined  y MIN_VALUE si tiene valor, se valida sobre ese, para escenario donde está indicado el rango mínimo pero no el máximo
                                            //Si solo se tiene MIN_VALUE y este es 0, se agrega escenario para validar que fieldValue sea exactamente igual
                                            if((parseFloat(fieldValue) > parseFloat(MIN_VALUE) && parseFloat(fieldValue) <= parseFloat(MAX_VALUE)) || (parseFloat(fieldValue) == parseFloat(MIN_VALUE) && parseFloat(MIN_VALUE) == 0) || (MAX_VALUE == undefined && ctcUtils.isStringNotEmpty(MIN_VALUE) && (parseFloat(fieldValue) > parseFloat(MIN_VALUE) && parseFloat(MIN_VALUE) != 0) || (parseFloat(MIN_VALUE) == 0 && parseFloat(MIN_VALUE) == parseFloat(fieldValue)))){
                                                //Se cumple el rango de la matriz de aprobación
                                                let currentApprovalMaxLevel = ctcUtils.app.approvalMaxLevel;
                                                //Guardamos el nivel de aprobación anterior al nuevo
                                                ctcUtils.app.approvalPreviousMaxLevel = currentApprovalMaxLevel;
                                                //Guardamos el nombre del flujo a ejecutar anterior al nuevo
                                                ctcUtils.app.approvalPreviousFlowName = ctcUtils.app.approvalFlowName;

                                                if(currentApprovalMaxLevel == undefined || HIERARCHY_LIST[i] >= currentApprovalMaxLevel){
                                                    //Si detectamos que el rango nuevo que se obtuvo tiene más jerarquía (mas niveles de aprobación), que el rango evaluado y guardado anteriormente, se sobrescribe el flujo actual por el de mayor jerarquía
                                                    //Guardamos el nombre del flujo/matriz que obtuvo el mayor nivel de jerarquía
                                                    ctcUtils.app.approvalmaxFlowName = APPROVAL_MATRIX_NAME;
                                                    //Se guarda el nivel máximo de aprobación a alcanzar en las aprobaciones
                                                    ctcUtils.app.approvalMaxLevel = HIERARCHY_LIST[i];
                                                    /**
                                                     * Se guarda el flujo de aprobaciones que se ejecutará al guardar la cotización
                                                     * Si la oportunidad es Cambio de Parámetros, se agrega la letra C al nombre del flujo
                                                    **/
                                                    ctcUtils.app.approvalFlowName = ctcUtils.app.opportunity.RecordType === 'RT_ChangeParameters'?FLOWS_LIST[i]+'C':FLOWS_LIST[i];
                                                    //Esto solo se ocupa para poder detonar el modal de solicitar aprobación, ELIMINAR cuando se implemente una lógica mejor en el modal
                                                    //ctcUtils.app.approvalCredito = 64;
                                                    ctcUtils.isAnApprovalRequired = true;
                                                }else if(ctcUtils.app.approvalmaxFlowName.startsWith(APPROVAL_MATRIX_NAME)){
                                                    //La variable maxFlowName guarda el nombre del flujo que tuve la mayor jerarquía, si se detecta que hubo un cambio en ese flujo y en vez de aumentar la jerarquía, la disminuyó, se hace una evalucación completa de todos las matrices y sus valores actuales
                                                    //Y se determina la nueva jerarquía
                                                    ctcUtils.app.approvalMaxLevel = null;
                                                    ctcUtils.app.approvalmaxFlowName = null;
                                                    ctcUtils.recalculateApprovalMatrix();
                                                }
                                            }
                                        }
                                        //Se calcula/obtiene el semáforo a mostrar en pospago 
                                        ctcUtils.getTrafficLightStyle(ctcUtils.app.approvalMaxLevel);
                                    }
                                }else if(APPROVALS_MATRIX != undefined && APPROVALS_MATRIX.length == 0 && !ctcUtils.doesWarningApprovalMatrixShowed){
                                    //El mensaje se muestra dos veces dado que por la forma en que está codificado todo, esta función se ejecuta dos veces al seleccionar el producto
                                    const WARNING_MESSAGE = 'No se ha configurado una matriz de aprobación para este producto, puede que las funciones de solicitud de aprobación no funcionen correctamente.';
                                    console.log(WARNING_MESSAGE);
                                    ctcUtils.showError(WARNING_MESSAGE,'Cuidado');
                                    ctcUtils.doesWarningApprovalMatrixShowed = true;
                                }
                                console.log('NIVEL MÁXIMO DE APROBACIÓN: '+ctcUtils.app.approvalMaxLevel);
                                console.log('NOMBRE DEL FLUJO DE APROBACIÓN: '+ctcUtils.app.approvalFlowName);
                            }catch(e){
                                const ERROR_MESSAGE = e.message;
                                console.error('Ha habido un error al evaluar las matrices de aprobación: '+ERROR_MESSAGE);
                                ctcUtils.showError(ERROR_MESSAGE,'Error');
                            }
                        }

                            * Obtiene la CSS class que se mostrará para el semáforo en pospago, validando el approvalLevel

                        ctcUtils.getTrafficLightStyle = function(approvalLevel){

                            }

                        /**
                            * Reevalúa todas las matrices de aprobación del producto actual, para obtener la nueva jerarquía
                            * 
                            * @returns void
                        */
                        ctcUtils.recalculateApprovalMatrix = function(){
                            const MATRIX_KEYS = Object.keys(ctcUtils.approvalMatrixFlows[ctcUtils.app.ProductCode])
                            for(let i=0;i<MATRIX_KEYS.length;i++){
                                //Si la matriz es de intereses y no hay cobro de intereses, se omite el evalúo
                                if((MATRIX_KEYS[i] == 'Intereses' && !ctcUtils.isNotInterestCharge) || MATRIX_KEYS[i] != 'Intereses' ){
                                    ctcUtils.evaluateTheApprovalStage(MATRIX_KEYS[i],ctcUtils.approvalMatrixFlows[ctcUtils.app.ProductCode][MATRIX_KEYS[i]]);
                                }
                            }
                        }

                                                    ComparadosLista.push(v1)
                                                }
                                            })

                                    console.log('[CEAM] Render 3: ', v);

                                    // v.ServiceDate = v.ServiceDate ? ctcUtils.parseDate(v.ServiceDate) : ''
                                    if(!cotizacion.idsProductos) {
                                        cotizacion.idsProductos = ''

                                        cotizacion.idsProductos += ', '
                                    }
                                    
                                    // nueva funcionalidad monoproducto calvarez 22_11_2017

                    vm.isAviableEFirma = function (objProduct){
                        var strCode = '';
                        var strFamily = '';
                        if(objProduct.Product2Id__o){
                            strCode = objProduct.Product2Id__o.ProductCode;
                            strFamily = objProduct.Product2Id__o.Family;
                        }
                        else if(objProduct.Product2){
                            strCode = objProduct.Product2.ProductCode;
                            strFamily = objProduct.Product2.Family;
                        }
                        var flag = false;
                        if(strFamily == 'Combustible' && (strCode == '30' || strCode == '30-TC4'))
                            flag = true;
                        else if(strFamily == 'Despensa' && strCode != '32-W' && strCode != '32-CH')
                            flag = true;
                        else if(strFamily == 'Regalo' && strCode != '48-W' && strCode != '48-CH')
                            flag = true;
                        else if (strFamily != 'Combustible' && strFamily != 'Despensa' && strFamily != 'Regalo')
                            flag = true;

                        return flag;
                    }
