    @TestVisible
    public static String init(String requestString) {
        SavePoint sp = DataBase.setSavepoint();
        String message ='';
        System.debug('requestString fe '+requestString);
        try {
            GLB_ClientReplicationWrapper contratoJSON = GLB_ClientReplicationWrapper.parse(requestString.Replace('Import','ImportSF'));

			    /* Televia Modificaciones */
            if(contratoJSON.Records == null || contratoJSON.Records.isEmpty() || String.isBlank(contratoJSON.Records[0].FlagTelevia)){
                successListglobal = new List<String>();
                DataBase.RollBack(sp);
                errorListglobal.add('Añadir valor en Televia ');
                message = createResponseBody(false, 400, new List<String>(), errorListglobal);
            }
            if( String.isBlank(contratoJSON.Records[0].PlasticOfBilling)){
                successListglobal = new List<String>();
                DataBase.RollBack(sp);
                errorListglobal.add('Añadir valor en PlasticOfBilling');
                message = createResponseBody(false, 400, new List<String>(), errorListglobal);
            } 
                /* Televia Modificaciones */
            CreateClients(contratoJSON);
            if(errorListglobal != null && !errorListglobal.IsEmpty()){
                successListglobal = new List<String>();
                DataBase.RollBack(sp);
                message = createResponseBody(true, 400, successListglobal, errorListglobal);
            }
            else{
                message = createResponseBody(true, 200, successListglobal, errorListglobal);
            }
        }
        catch (Exception e){
            DataBase.RollBack(sp);
            errorListglobal.add('Error al crear Contrato: ' + e.getMessage());
            message = createResponseBody(false, 400, new List<String>(), errorListglobal);
        }
        return message;
    }
    private static Boolean validateIfClientIsAlreadyMigrated(List<GLB_ClientReplicationWrapper.Records> records) {
        Boolean existing = false;
        Set<String> recordAS400Ids = new Set<String>();
        List<ContratoFilial__c> cfList = new List<ContratoFilial__c>();
        alreadyMigratedCF = new ContratoFilial__c();
        for(GLB_ClientReplicationWrapper.Records record : records) {
            recordAS400Ids.add(record.LegalEntityCode);
        }
        cfList = [
            SELECT Id, Name, GLB_MigratedAs400ClientId__c
            FROM ContratoFilial__c
            WHERE GLB_MigratedAs400ClientId__c IN :recordAS400Ids

 	@TestVisible
    private static Map<String,ContratoFilial__c> CreateClients(GLB_ClientReplicationWrapper payload){
        List<GLB_ClientReplicationWrapper.Records> records = (payload != null && payload.Records != null) ? payload.Records : new List<GLB_ClientReplicationWrapper.Records>();
        if(validateIfClientIsAlreadyMigrated(records)) {
            String errorMessage = 'ERROR - El cliente (' + alreadyMigratedCF.GLB_MigratedAs400ClientId__c + ') ya ha sido migrado - Id: ' + alreadyMigratedCF.Name;
            errorListglobal.add(errorMessage);
            return null;
        }
        //Cuentas - TODO: Check some fields
        Map<String,Account> mapCuentas = creaCuentas(records);
        //Direcciones
        Map<String,CatalogoDirecciones__c> mapDirecciones = creaDirecciones(records,mapCuentas);
        //Entidades Cuenta y Entidades legales
        Entidades entidadesLeg = creaEntidadesCuentaLegales(records,mapCuentas,mapDirecciones);
        //Contactos
        Map<String,Contact> mapContacts = creaContactos(records,mapCuentas);
        //Oportunidades
        Map<String,Opportunity> mapOpps = creaOpps(records,mapCuentas,mapContacts,entidadesLeg);
        //Cotizaciones
        Map<String,Quote> MapQuotes = creaQuotes(records, mapCuentas, mapOpps);
        //Partidas de cotizaciones
        Map<String,QuoteLineItem> mapQuotesLine = creaQuoteLineItems(records,mapCuentas,mapQuotes,mapOpps);
        //Terminos comerciales
        Map<String,ProductoConcepto__c> mapComercialTerms = creaComercialTerms(records,mapQuotesLine);
        // modificaciones Fix Name Migracion
        // checksOpps
        // NOTE: No hacemos query adicional a Opportunity para obtener el nombre.
        // El cálculo del firstPart se deriva del Name que ya construimos en creaOpps.
        // (Evita consumir la query #101 cuando ya vienen límites altos por automatización externa)
        checkOppsEmptyContracts(mapOpps);

        // Grupos de facturación (nueva funcionalidad)
        // Solo se ejecuta cuando el contrato viene configurado para facturación por grupo.
        Boolean ejecutarGpoFact = false;
        for (GLB_ClientReplicationWrapper.Records r : records) {
            if (r == null) continue;
            String comoFactura = normaliceString(r.CompanyInvoices).toUpperCase();
            if (String.isNotBlank(comoFactura) && comoFactura.trim() == 'POR GRUPO DE FACTURACION') {
                ejecutarGpoFact = true;
                break;
            }
            // fallback: algunos payloads usan el campo dentro de Contracts.HowToBill
            if (r.Contracts != null) {
                for (GLB_ClientReplicationWrapper.Contracts c : r.Contracts) {
                    if (c == null) continue;
                    String comoFacturaBill = normaliceString(c.HowToBill).toUpperCase();
                    if (String.isNotBlank(comoFacturaBill) && comoFacturaBill.trim() == 'POR GRUPO DE FACTURACION') {
                        ejecutarGpoFact = true;
                        break;
                    }
                }
                if (ejecutarGpoFact) break;
            }
        }

          //Administracion de ventas -> GLB_MigrationToMercedes__c
        Map<String,Contrato2__c> adminVent = creaAdministracionVenta(records,mapCuentas,mapDirecciones,mapContacts,entidadesLeg,mapQuotesLine,mapComercialTerms);
        //Contratos filiales
        Map<String,ContratoFilial__c> contrFilial = creaContratosFiliales(records,mapCuentas,mapOpps,mapDirecciones,mapContacts,entidadesLeg,mapQuotesLine,mapComercialTerms,adminVent);

        if (ejecutarGpoFact) {
            List<Grupo_de_facturacion__c> grupos = processBillingGroups(payload, mapCuentas, mapOpps, entidadesLeg);
            enqueueBillingGroupsPhantom(payload, grupos, contrFilial, mapOpps);
        }

        // Ejecutar el update consolidado de Opportunity (SyncedQuoteId y SubCuentaMigracion__c) una sola vez.
        flushPendingOppUpdates();

        // call async process
        if(contrFilial != null) {
            for(String key : adminVent.keyset()){
                System.enqueueJob(new AsyncExecutionExample(adminVent.get(key), mapOpps.get(key), key));
            }
        }
        return null;
    }

    public static void checkOppsEmptyContracts(Map<String,Opportunity> mapOpps){
        if(mapOpps == null || mapOpps.isEmpty()){
            return;
        }

        // Para compatibilidad con el comportamiento actual (payload típico 1 registro),
        // tomamos el firstPart del primer elemento.
        for(Opportunity oppItem : mapOpps.values()){
            if(String.isNotBlank(oppItem.Name) && oppItem.Name.contains('-')){
                firstPart = oppItem.Name.split('-')[0];
            } else {
                firstPart = String.isNotBlank(oppItem.Name) ? oppItem.Name : '';
            }
            break;
        }
    }

    @TestVisible
    // Consolidación de updates a Opportunity para evitar múltiples DML (y automatizaciones repetidas).
    private static Map<Id, Opportunity> pendingOppUpdates = new Map<Id, Opportunity>();

    private static void upsertPendingOppUpdates(List<Opportunity> oppUpdates){
        if(oppUpdates == null || oppUpdates.isEmpty()){
            return;
        }
        for(Opportunity o : oppUpdates){
            if(o == null || o.Id == null) continue;
            if(!pendingOppUpdates.containsKey(o.Id)){
                pendingOppUpdates.put(o.Id, new Opportunity(Id = o.Id));
            }
            Opportunity merged = pendingOppUpdates.get(o.Id);

            // Merge de campos (solo asignamos si viene valor)
            if(o.SyncedQuoteId != null) merged.SyncedQuoteId = o.SyncedQuoteId;
            if(o.SubCuentaMigracion__c != null) merged.SubCuentaMigracion__c = o.SubCuentaMigracion__c;
        }
    }

    private static void flushPendingOppUpdates(){
        if(pendingOppUpdates == null || pendingOppUpdates.isEmpty()){
            return;
        }
        Database.SaveResult[] results = Database.update(pendingOppUpdates.values(), false);
        List<Opportunity> updatedList = pendingOppUpdates.values();
        for(Integer i = 0; i < results.size(); i++){
            if(results[i].isSuccess()){
                successListglobal.add(updatedList[i].Id + ': ' + results[i].getId());
            }else{
                for(Database.Error err : results[i].getErrors()){
                    errorListglobal.add(err.getMessage());
                }
            }
        }
        pendingOppUpdates.clear();
    }

    private static String createResponseBody(Boolean succeeded, Integer code, List<String> successList, List<String> errorList) {
        Response res = new Response();
        res.code = code;
        res.succeeded = succeeded;
        res.errors = null;
        res.success = null;
        if(successList.size() > 0) {
            List<Success> successsList = new List<Success>();
            for(String successString : successList) {
                Success success = new Success();
                success.description = successString;
                successsList.add(success);
            }
            res.success = successsList;
        }
        if(errorList.size() > 0) {
            List<Errors> errorsList = new List<Errors>();
            for(String errorString : errorList) {
                Errors error = new Errors();
                error.description = errorString;
                errorsList.add(error);
            }
            res.errors = errorsList;
        }
        String jsonResponse = System.JSON.serialize(res);
        return jsonResponse;
    }

    public static Date reformatDate(String dateInput){
        // Date res;
        List<String> dateSplitted = new List<String>();
        String newDateString;
        if (dateInput == null) {
			return null;
		}
        dateSplitted = dateInput.split('/'); // "29/10/2020" -> "29", "10", "2020"

        String day = '';
		if (toDateTime) {
			year = fechaOriginal.year();
            mes = String.ValueOf(fechaOriginal.Month() + 1);
			month = '0' + mes;
            dia = String.ValueOf(fechaOriginal.Day());
			day = '0' + dia;
			nuevaFecha = zeroHour ? year + '-' + month + '-' + day + ' 00:00:00Z' : year + '-' + month + '-' + day + ' 23:59:00Z';
			return nuevaFecha;
		} else {
			year = fechaOriginal.year();
			mes = String.ValueOf(fechaOriginal.Month() + 1);
			month = '0' + mes;
			dia = String.ValueOf(fechaOriginal.Day());
			day = '0' + dia;
			nuevaFecha = year + '-' + month + '-' + day;
			return nuevaFecha;
		}
	}

     @TestVisible
    private static List<Grupo_de_facturacion__c> processBillingGroups(
        GLB_ClientReplicationWrapper payload,
        Map<String, Account> mapCuentas,
        Map<String, Opportunity> mapOpps,
        Entidades entidadesLeg
    ) {
        if (payload == null) return new List<Grupo_de_facturacion__c>();
        if (payload.Records == null || payload.Records.isEmpty()) return new List<Grupo_de_facturacion__c>();

        // Acumular listas desde los Records (por contrato del WS)
        List<GLB_ClientReplicationWrapper.BillingGroupStructure> allGroups = new List<GLB_ClientReplicationWrapper.BillingGroupStructure>();

        for (GLB_ClientReplicationWrapper.Records r : payload.Records) {
            if (r == null) continue;
            if (r.BillingGroupStructure != null) allGroups.addAll(r.BillingGroupStructure);
        }

        if (allGroups.isEmpty() && entidadesLeg.entidadesLegales.isEmpty()) {
            return new List<Grupo_de_facturacion__c>();
        }

        GLB_ClientReplicationWrapper.Records recordPrincipal = payload.Records[0];
        if (recordPrincipal == null || String.isBlank(recordPrincipal.RFC)) {
            return new List<Grupo_de_facturacion__c>();
        }

        // Cuenta para direcciones (según indicación: desde EntidadCuenta / ContratoFilial)
        Id cuentaId = null;
        if (entidadesLeg != null && entidadesLeg.entidadesCuenta != null && entidadesLeg.entidadesCuenta.containsKey(recordPrincipal.RFC)) {
            cuentaId = entidadesLeg.entidadesCuenta.get(recordPrincipal.RFC).Cuenta__c;
        }
        if (cuentaId == null && mapCuentas != null && mapCuentas.containsKey(recordPrincipal.RFC)) {
            cuentaId = mapCuentas.get(recordPrincipal.RFC).Id;
        }
        if (cuentaId == null) {
            return new List<Grupo_de_facturacion__c>();
        }

        // Opportunity para relacionar grupos
        Id oppId = null;
        if (mapOpps != null && mapOpps.containsKey(recordPrincipal.LegalEntityCode)) {
            oppId = mapOpps.get(recordPrincipal.LegalEntityCode).Id;
        }
        // 4) Grupo_de_facturacion__c con flags compatibles (lote/activo)
        Integer lote = 1;
        List<Grupo_de_facturacion__c> last = [
            SELECT Id, NTT_Lote__c
            FROM Grupo_de_facturacion__c
            WHERE Account__c = :cuentaId
            ORDER BY NTT_Lote__c DESC NULLS LAST, CreatedDate DESC
            LIMIT 1
        ];
        if (!last.isEmpty() && last[0].NTT_Lote__c != null) {
            lote = Integer.valueOf(last[0].NTT_Lote__c);
        }

        List<Grupo_de_facturacion__c> gruposToInsert = new List<Grupo_de_facturacion__c>();
        if (!allGroups.isEmpty()) {
            for (GLB_ClientReplicationWrapper.BillingGroupStructure bg : allGroups) {
                if (bg == null) continue;

                Grupo_de_facturacion__c g = new Grupo_de_facturacion__c();
                g.Name = bg.BillingGroupName;
                g.IDMigracionContrato__c = String.valueOf(bg.IDContractWattsMother);
                g.CodigoAS400MIgracion__c = bg.LegalEntityCodeMother;
                g.Account__c = cuentaId;
                g.Opportunity__c = oppId;
                g.RFC_Text__c = bg.RFCGrupo;
                g.Centro_de_Costos__c = String.valueOf(bg.GroupIdtoBilling);
                g.IsPrincipalMigration__c = bg.ISMother;
                if (entidadesLeg.entidadesLegales.containsKey(bg.RFCGrupo)){
                    g.Entidad_Legal__c = entidadesLeg.entidadesLegales.get(bg.RFCGrupo).Id;
                }

                g.Activo__c = true;
                g.NTT_Lote__c = lote;
                g.NTT_Es_Ultimo_Lote__c = true;

                gruposToInsert.add(g);
            }
        }
        return gruposToInsert;
    }

    @TestVisible
    private static void enqueueBillingGroupsPhantom(
        GLB_ClientReplicationWrapper payload,
        List<Grupo_de_facturacion__c> grupos,
        Map<String, ContratoFilial__c> contratosFiliales,
        Map<String, Opportunity> mapOpps
    ) {
        if (payload == null || payload.Records == null || payload.Records.isEmpty()
            || grupos == null || grupos.isEmpty() || contratosFiliales == null) {
            return;
        }

        GLB_ClientReplicationWrapper.Records principal = payload.Records[0];
        if (principal == null || String.isBlank(principal.LegalEntityCode)
            || !contratosFiliales.containsKey(principal.LegalEntityCode)) {
            return;
        }

        ContratoFilial__c contratoFilial = contratosFiliales.get(principal.LegalEntityCode);
        Id opportunityId = mapOpps != null && mapOpps.containsKey(principal.LegalEntityCode)
            ? mapOpps.get(principal.LegalEntityCode).Id
            : null;

        System.enqueueJob(new NTTBillingGroupsPhantomQueueable(
            grupos,
            contratoFilial.Id,
            opportunityId,
            contratoFilial.GLB_ID_Cuenta__c
        ));
    }

    public class NTTBillingGroupsPhantomQueueable implements Queueable, Database.AllowsCallouts {
        private List<Grupo_de_facturacion__c> grupos;
        private Id contratoFilialId;
        private Id opportunityId;
        private Double glbIdCuentaActual;

        public NTTBillingGroupsPhantomQueueable(
            List<Grupo_de_facturacion__c> grupos,
            Id contratoFilialId,
            Id opportunityId,
            Decimal glbIdCuentaActual
        ) {
            this.grupos = grupos;
            this.contratoFilialId = contratoFilialId;
            this.opportunityId = opportunityId;
            this.glbIdCuentaActual = glbIdCuentaActual == null
                ? null
                : Double.valueOf(String.valueOf(glbIdCuentaActual));
        }

        public void execute(QueueableContext context) {
            NTTPhantomGrupoFacturacionService.Request request =
                new NTTPhantomGrupoFacturacionService.Request();
            request.grupos = grupos;
            request.contratoFilialId = contratoFilialId;
            request.opportunityId = opportunityId;
            request.glbIdCuentaActual = glbIdCuentaActual;

            List<NTTPhantomGrupoFacturacionService.Response> responses =
                NTTPhantomGrupoFacturacionService.generarPhantomYGuardar(
                    new List<NTTPhantomGrupoFacturacionService.Request>{ request }
                );

            if (responses != null && !responses.isEmpty() && !responses[0].success) {
                System.debug(LoggingLevel.ERROR,
                    'Error al generar grupos de facturación en Phantom: ' + responses[0].message);
            }
        }
    }
